```python
def init_mapping(self):
    """
    DAG Slack-Weighted Tension Network Spring Relaxation (DSTWNSR)

    Models qubits as particles on the hardware graph connected by springs.
    Spring stiffness = gate_weight / (slack + 1), where:
      - gate_weight = remaining schedule length from that gate (max_t - earliest + 1)
      - slack       = latest_start - earliest_start  (0 on critical path)

    Critical-path gates create stiff springs → pull their qubits close together
    on the hardware topology.  Springs are relaxed via:
      1. Greedy anchor-and-expand placement (most-connected qubit first).
      2. Iterative pairwise swap local search until no improving swap exists.
    """
    from collections import defaultdict, deque

    # ------------------------------------------------------------------ #
    # 0. Collect logical qubits / guard for empty circuits                #
    # ------------------------------------------------------------------ #
    logical_qubit_set = set()
    for qubits in self.access.values():
        logical_qubit_set.update(qubits)
    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    mapping_dict         = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    if not logical_qubits:
        self.mapping_dict         = mapping_dict
        self.reverse_mapping_dict = reverse_mapping_dict
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # ------------------------------------------------------------------ #
    # 1. Build a lightweight sequential 2Q-gate DAG for slack computation #
    #    (per-qubit last-gate tracking  →  O(|gates|) construction)       #
    # ------------------------------------------------------------------ #
    two_qubit_gates = {g: (q[0], q[1]) for g, q in self.access.items() if len(q) == 2}

    if not two_qubit_gates:
        # No 2Q interactions: trivial identity mapping
        self.mapping_dict         = mapping_dict
        self.reverse_mapping_dict = reverse_mapping_dict
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    sorted_2q  = sorted(two_qubit_gates.keys())
    dag_succ   = defaultdict(set)
    dag_pred   = defaultdict(set)
    last_on_q  = {}   # qubit -> most recent gate that touched it

    for g in sorted_2q:
        q1, q2 = two_qubit_gates[g]
        for q in (q1, q2):
            if q in last_on_q:
                pred = last_on_q[q]
                dag_succ[pred].add(g)
                dag_pred[g].add(pred)
            last_on_q[q] = g

    # ------------------------------------------------------------------ #
    # 2. Topological sort + earliest start (forward pass)                 #
    # ------------------------------------------------------------------ #
    in_deg   = {g: len(dag_pred[g]) for g in sorted_2q}
    earliest = {g: 0 for g in sorted_2q}
    topo_order = []
    queue = deque(g for g in sorted_2q if in_deg[g] == 0)

    while queue:
        g = queue.popleft()
        topo_order.append(g)
        for s in dag_succ[g]:
            earliest[s] = max(earliest[s], earliest[g] + 1)
            in_deg[s] -= 1
            if in_deg[s] == 0:
                queue.append(s)

    # ------------------------------------------------------------------ #
    # 3. Latest start (backward pass) → slack per gate                   #
    # ------------------------------------------------------------------ #
    max_t  = max(earliest.values(), default=0)
    latest = {g: max_t for g in sorted_2q}

    for g in reversed(topo_order):
        for s in dag_succ[g]:
            latest[g] = min(latest[g], latest[s] - 1)

    # slack[g] = 0  →  gate is on the critical path (stiffest spring)
    slack = {g: max(0, latest[g] - earliest[g]) for g in sorted_2q}

    # ------------------------------------------------------------------ #
    # 4. Build spring stiffness per logical qubit pair                    #
    #                                                                      #
    #    stiffness = gate_weight / (slack + 1)                            #
    #    gate_weight = max_t - earliest[g] + 1   (remaining work)        #
    # ------------------------------------------------------------------ #
    spring_stiffness = defaultdict(float)   # (q_lo, q_hi) -> total stiffness
    raw_interactions = defaultdict(float)   # same key  -> unweighted count

    for g, (q1, q2) in two_qubit_gates.items():
        gate_weight = max_t - earliest[g] + 1
        k = gate_weight / (slack[g] + 1.0)
        key = (min(q1, q2), max(q1, q2))
        spring_stiffness[key]  += k
        raw_interactions[key]  += 1.0

    # Adjacency with spring stiffness as edge weights
    spring_nbrs = defaultdict(lambda: defaultdict(float))
    for (q1, q2), k in spring_stiffness.items():
        spring_nbrs[q1][q2] += k
        spring_nbrs[q2][q1] += k

    total_spring = {q: sum(spring_nbrs[q].values()) for q in logical_qubits}

    # ------------------------------------------------------------------ #
    # 5. Hardware anchor: physical node with minimum sum of distances     #
    #    (hardware centroid — most globally accessible node)              #
    # ------------------------------------------------------------------ #
    center_phys = min(
        physical_qubits,
        key=lambda p: sum(
            self.distance_matrix[p][o]
            for o in physical_qubits
            if self.distance_matrix[p][o] != float('inf')
        )
    )

    # ------------------------------------------------------------------ #
    # 6. Greedy spring-energy placement                                   #
    #                                                                      #
    #    Order: most-connected (highest Σ stiffness) qubit first.        #
    #    At each step assign the unplaced qubit to the physical node      #
    #    minimising Σ_{already placed q'} k(q,q') * dist(p, phys(q')).  #
    # ------------------------------------------------------------------ #
    sorted_logical = sorted(logical_qubits, key=lambda q: -total_spring.get(q, 0))

    lq_to_phys  = {}
    placed_phys = set()

    for lq in sorted_logical:
        if not lq_to_phys:
            # Anchor: highest-stiffness qubit goes to hardware centroid
            lq_to_phys[lq] = center_phys
            placed_phys.add(center_phys)
            continue

        # Candidate physical positions (hardware neighbours of placed set first)
        candidates = [
            nb
            for p in placed_phys
            for nb in self.backend[p]
            if nb not in placed_phys
        ]
        if not candidates:
            candidates = [p for p in physical_qubits if p not in placed_phys]
        if not candidates:
            break

        # Spring energy at candidate position p
        def spring_energy(p, _lq=lq):
            e = 0.0
            for lq2, p2 in lq_to_phys.items():
                k = spring_nbrs[_lq].get(lq2, 0.0)
                if k == 0.0:
                    continue
                d = self.distance_matrix[p][p2]
                e += k * (d if d != float('inf') else 1e9)
            return e

        best_phys = min(candidates, key=spring_energy)
        lq_to_phys[lq] = best_phys
        placed_phys.add(best_phys)

    # ------------------------------------------------------------------ #
    # 7. Spring relaxation via iterative pairwise swap local search       #
    #                                                                      #
    #    Swap lq1 ↔ lq2 whenever Δ(total spring energy) < 0.             #
    #    Uses raw interaction_neighbors as weights (symmetrises stiffness #
    #    to avoid double-counting in the delta formula).                  #
    # ------------------------------------------------------------------ #
    lq_list = [lq for lq in logical_qubits if lq in lq_to_phys]
    n = len(lq_list)

    for _pass in range(30):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lq1, lq2 = lq_list[i], lq_list[j]
                p1, p2   = lq_to_phys[lq1], lq_to_phys[lq2]

                delta = 0.0
                # Energy change for lq1's springs (excluding the lq1↔lq2 spring)
                for lq3, k in spring_nbrs[lq1].items():
                    if lq3 not in lq_to_phys or lq3 == lq2:
                        continue
                    p3    = lq_to_phys[lq3]
                    d_old = self.distance_matrix[p1][p3]
                    d_new = self.distance_matrix[p2][p3]
                    delta += k * (
                        (d_new if d_new != float('inf') else 1e9) -
                        (d_old if d_old != float('inf') else 1e9)
                    )

                # Energy change for lq2's springs (excluding the lq1↔lq2 spring)
                for lq3, k in spring_nbrs[lq2].items():
                    if lq3 not in lq_to_phys or lq3 == lq1:
                        continue
                    p3    = lq_to_phys[lq3]
                    d_old = self.distance_matrix[p2][p3]
                    d_new = self.distance_matrix[p1][p3]
                    delta += k * (
                        (d_new if d_new != float('inf') else 1e9) -
                        (d_old if d_old != float('inf') else 1e9)
                    )

                # Note: the lq1↔lq2 spring energy is invariant under their swap
                # because dist(p1,p2) = dist(p2,p1) — no contribution to delta.

                if delta < -1e-9:
                    lq_to_phys[lq1], lq_to_phys[lq2] = p2, p1
                    improved = True

        if not improved:
            break

    # ------------------------------------------------------------------ #
    # 8. Commit: build strict 1-to-1 bijection via in-place swap         #
    # ------------------------------------------------------------------ #
    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]

        mapping_dict[lq]                    = target_phys
        mapping_dict[displaced_lq]          = current_phys
        reverse_mapping_dict[target_phys]   = lq
        reverse_mapping_dict[current_phys]  = displaced_lq

    self.mapping_dict         = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

---

**How it implements DSTWNSR, step by step:**

| Phase | What it does | DSTWNSR concept |
|---|---|---|
| **DAG construction** | Builds per-qubit sequential dependency graph in O(|gates|) | "Gate dependencies create springs" |
| **Slack computation** | Forward pass → `earliest[g]`, backward pass → `latest[g]`, `slack = latest - earliest` | "Inverse slack in the DAG schedule" |
| **Spring stiffness** | `k = (max_t - earliest + 1) / (slack + 1)` accumulated per qubit pair | "Spring stiffness ∝ gate_weight × inverse slack" |
| **Anchor** | Most-spring-connected qubit → hardware centroid | Initialize particles on the hardware graph |
| **Greedy placement** | Each subsequent qubit placed at the hardware neighbor minimizing `Σ k·dist` | "Springs pull particles together on hardware topology" |
| **Relaxation** | 30-pass pairwise swap local search minimizing total `Σ k·dist(phys(q1), phys(q2))` | "Springs relax on hardware graph topology" |
| **Bijection commit** | In-place swap pattern guarantees strict 1-to-1 mapping | Validity guarantee |

**Key design choices:**
- `gate_weight = max_t - earliest + 1`: gates earlier in the schedule have more remaining work → stronger pull
- `slack = 0` (critical path) → `stiffness = gate_weight` (maximum)
- `slack > 0` → stiffness is dampened proportionally → non-critical interactions create softer springs
- The direct spring between `lq1` and `lq2` is correctly excluded from the swap delta (its distance is invariant under their mutual swap)
