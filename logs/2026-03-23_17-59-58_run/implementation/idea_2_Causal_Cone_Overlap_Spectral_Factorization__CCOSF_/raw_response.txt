def init_mapping(self):
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    from collections import defaultdict, deque

    # ── 1. Collect all logical qubits used in the circuit ──────────────────
    logical_qubits = sorted({q for qubits in self.access.values() for q in qubits})
    num_logical = len(logical_qubits)

    if num_logical == 0:
        self.mapping_dict = {}
        self.reverse_mapping_dict = {}
        return

    # ── 2. Per-qubit ordered gate list (program order via sorted gate IDs) ─
    qubit_gate_list = defaultdict(list)
    for gate in sorted(self.access.keys()):
        for q in self.access[gate]:
            qubit_gate_list[q].append(gate)

    # ── 3. Build gate-level DAG from qubit-sequential dependencies ─────────
    gates = list(self.access.keys())
    successors  = defaultdict(set)
    predecessors = defaultdict(set)
    for q_gates in qubit_gate_list.values():
        for i in range(len(q_gates) - 1):
            g_prev, g_next = q_gates[i], q_gates[i + 1]
            successors[g_prev].add(g_next)
            predecessors[g_next].add(g_prev)

    # ── 4. Topological sort (Kahn's algorithm) ─────────────────────────────
    temp_deg = {g: len(predecessors[g]) for g in gates}
    queue = deque(g for g in gates if temp_deg[g] == 0)
    topo_order = []
    while queue:
        g = queue.popleft()
        topo_order.append(g)
        for s in successors[g]:
            temp_deg[s] -= 1
            if temp_deg[s] == 0:
                queue.append(s)

    # ── 5. Forward cones via DP in reverse topological order ───────────────
    # FC[g] = set of all gates reachable forward from g (g included)
    FC = {g: {g} for g in gates}
    for g in reversed(topo_order):
        for s in successors[g]:
            FC[g] |= FC[s]

    # ── 6. Backward cones via DP in topological order ──────────────────────
    # BC[g] = set of all gates that can reach g backward (g included)
    BC = {g: {g} for g in gates}
    for g in topo_order:
        for p in predecessors[g]:
            BC[g] |= BC[p]

    # ── 7. Aggregate cones at qubit level ──────────────────────────────────
    qubit_FC = {}
    qubit_BC = {}
    for q in logical_qubits:
        qg = qubit_gate_list[q]
        if qg:
            qubit_FC[q] = set().union(*(FC[g] for g in qg))
            qubit_BC[q] = set().union(*(BC[g] for g in qg))
        else:
            qubit_FC[q] = set()
            qubit_BC[q] = set()

    # ── 8. Build symmetric Cone Overlap Matrix ─────────────────────────────
    # C[i][j] = |FC(qi) ∩ FC(qj)| + |BC(qi) ∩ BC(qj)|
    C = np.zeros((num_logical, num_logical))
    for i in range(num_logical):
        qi = logical_qubits[i]
        for j in range(i + 1, num_logical):
            qj = logical_qubits[j]
            val = float(
                len(qubit_FC[qi] & qubit_FC[qj]) +
                len(qubit_BC[qi] & qubit_BC[qj])
            )
            C[i, j] = val
            C[j, i] = val

    # ── 9. Spectral embedding of logical qubits ────────────────────────────
    # Eigenvectors of C corresponding to the largest eigenvalues embed
    # qubits with high mutual cone overlap close together in R^k.
    k = min(3, num_logical)
    try:
        evals, evecs = np.linalg.eigh(C)          # eigh: symmetric, returns sorted ascending
        evecs = evecs[:, np.argsort(evals)[::-1]]  # re-sort descending
        logical_emb = evecs[:, :k]                 # (num_logical, k)
    except np.linalg.LinAlgError:
        logical_emb = np.eye(num_logical, k)

    # ── 10. Spectral embedding of physical qubits ──────────────────────────
    # Build a hardware affinity matrix H[i][j] = 1/dist(pi, pj)
    # Eigenvectors of H give a spatial layout of the hardware.
    phys_qubits = sorted(self.distance_matrix.keys())
    num_phys    = len(phys_qubits)
    pidx        = {p: i for i, p in enumerate(phys_qubits)}

    H = np.zeros((num_phys, num_phys))
    for pi in phys_qubits:
        for pj, d in self.distance_matrix[pi].items():
            if pj in pidx and d > 0:
                H[pidx[pi], pidx[pj]] = 1.0 / d

    try:
        h_evals, h_evecs = np.linalg.eigh(H)
        h_evecs  = h_evecs[:, np.argsort(h_evals)[::-1]]
        phys_emb = h_evecs[:, :k]                  # (num_phys, k)
    except np.linalg.LinAlgError:
        phys_emb = np.eye(num_phys, k)

    # Ensure both embeddings have exactly k columns
    def _pad(arr, cols):
        if arr.shape[1] < cols:
            return np.hstack([arr, np.zeros((arr.shape[0], cols - arr.shape[1]))])
        return arr[:, :cols]

    logical_emb = _pad(logical_emb, k)
    phys_emb    = _pad(phys_emb,    k)

    # ── 11. Optimal 1-to-1 assignment via Hungarian algorithm ──────────────
    # cost[i][j] = Euclidean distance between logical qubit i and physical j
    # in the shared spectral space.
    cost = np.linalg.norm(logical_emb[:, None, :] - phys_emb[None, :, :], axis=2)
    # Guarantee validity: linear_sum_assignment yields a strict 1-to-1 match.
    row_ind, col_ind = linear_sum_assignment(cost)

    self.mapping_dict         = {}
    self.reverse_mapping_dict = {}
    for li, pi in zip(row_ind, col_ind):
        lq = logical_qubits[li]
        pq = phys_qubits[pi]
        self.mapping_dict[lq]         = pq
        self.reverse_mapping_dict[pq] = lq

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

**Design walkthrough:**

| Step | What happens | Why |
|------|-------------|-----|
| **DAG construction** | Sequential per-qubit dependencies → edges | Captures true program order without calling `generate_dag` (not yet available at `init_mapping` time) |
| **Forward cone FC[g]** | DP in reverse topo order: `FC[g] = {g} ∪ FC[succ]` | All gates g can transitively influence |
| **Backward cone BC[g]** | DP in topo order: `BC[g] = {g} ∪ BC[pred]` | All gates that can influence g |
| **Qubit cones** | `qubit_FC[q]` = union of FC over all gates touching q | Qubit-level influence horizon |
| **Overlap matrix C** | `C[i,j] = \|FC∩FC\| + \|BC∩BC\|` | High overlap → must be spatially close |
| **Spectral embed (logical)** | Top-k eigenvectors of C | Highly correlated qubits land nearby in R^k |
| **Spectral embed (physical)** | Top-k eigenvectors of hardware affinity `H[i,j]=1/dist` | Nearby hardware qubits land nearby in same R^k |
| **Hungarian matching** | `linear_sum_assignment(‖logical_emb - phys_emb‖)` | Globally optimal, strictly 1-to-1, O(n³) |

**Validity guarantees:**
- `linear_sum_assignment` always returns a perfect matching when `num_logical ≤ num_phys`, so every logical qubit gets exactly one unique physical qubit.
- `eigh` is used (not `eig`) for the symmetric matrices, giving real, stable eigenvalues.
- `_pad` handles edge cases where `num_logical < k` or `num_phys < k`.
