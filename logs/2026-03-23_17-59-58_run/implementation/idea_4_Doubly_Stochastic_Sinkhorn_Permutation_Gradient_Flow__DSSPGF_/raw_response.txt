Now I have full clarity on the data structures. The `distance_matrix` is a 2D list indexed by physical qubit ID, and `self.backend` is an adjacency dict whose keys are the canonical physical qubit IDs.

Here is the complete `init_mapping` implementation:

```python
def init_mapping(self):
    import numpy as np
    from scipy.optimize import linear_sum_assignment

    # ------------------------------------------------------------------ #
    # 0. Collect logical and physical qubit sets                           #
    # ------------------------------------------------------------------ #
    logical_qubits = sorted(set(q for qubits in self.access.values() for q in qubits))
    n_l = len(logical_qubits)
    phys_qubits   = sorted(self.backend.keys())   # canonical IDs from the hardware graph
    n_p           = len(phys_qubits)

    # Trivial fallback – should never be reached on a real circuit
    if n_l == 0 or n_p == 0:
        self.mapping_dict         = {}
        self.reverse_mapping_dict = {}
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    log_idx  = {lq: i for i, lq in enumerate(logical_qubits)}
    phys_idx = {pq: i for i, pq in enumerate(phys_qubits)}

    # ------------------------------------------------------------------ #
    # 1. Build logical interaction matrix  C  (n_l × n_l)                 #
    #    C[i,j] = number of 2-qubit gates between logical qubits i and j  #
    # ------------------------------------------------------------------ #
    C = np.zeros((n_l, n_l), dtype=float)
    for qubits in self.access.values():
        if len(qubits) == 2:
            i, j = log_idx[qubits[0]], log_idx[qubits[1]]
            C[i, j] += 1.0
            C[j, i] += 1.0

    # ------------------------------------------------------------------ #
    # 2. Build physical distance matrix  D  (n_p × n_p)                   #
    # ------------------------------------------------------------------ #
    D = np.zeros((n_p, n_p), dtype=float)
    for pi_id in phys_qubits:
        pi = phys_idx[pi_id]
        for pj_id in phys_qubits:
            pj = phys_idx[pj_id]
            if pi_id != pj_id:
                d = self.distance_matrix[pi_id][pj_id]
                D[pi, pj] = d if d != float('inf') else n_p

    # ------------------------------------------------------------------ #
    # 3. Embed into a square n_p × n_p system                             #
    #    (pad C with zeros for dummy logical rows/cols when n_l < n_p)    #
    # ------------------------------------------------------------------ #
    n     = n_p
    C_pad = np.zeros((n, n), dtype=float)
    C_pad[:n_l, :n_l] = C

    # ------------------------------------------------------------------ #
    # 4. Initialise P as a random doubly stochastic matrix via Sinkhorn   #
    # ------------------------------------------------------------------ #
    def sinkhorn(M, n_iter=40, eps=1e-12):
        """Project M onto the Birkhoff polytope (doubly stochastic matrices)."""
        for _ in range(n_iter):
            row_sums = M.sum(axis=1, keepdims=True)
            M = M / np.where(row_sums > eps, row_sums, eps)
            col_sums = M.sum(axis=0, keepdims=True)
            M = M / np.where(col_sums > eps, col_sums, eps)
        return M

    rng = np.random.RandomState(42)
    P   = np.abs(rng.randn(n, n)) + 1e-3
    P   = sinkhorn(P)

    # ------------------------------------------------------------------ #
    # 5. Gradient flow on the Birkhoff polytope                           #
    #                                                                      #
    #    Cost:     f(P) = trace(P^T C_pad P D)                            #
    #    Gradient: ∂f/∂P = 2 · C_pad · P · D   (C_pad, D symmetric)      #
    #                                                                      #
    #    After each gradient step, clip negatives and re-project via       #
    #    Sinkhorn to stay in the doubly stochastic polytope.               #
    # ------------------------------------------------------------------ #
    lr       = 0.5
    lr_decay = 0.99
    n_steps  = 200

    for _ in range(n_steps):
        grad = 2.0 * (C_pad @ P @ D)   # (n × n)
        P    = P - lr * grad
        P    = np.clip(P, 1e-10, None)
        P    = sinkhorn(P, n_iter=15)
        lr  *= lr_decay

    # ------------------------------------------------------------------ #
    # 6. Round the converged soft assignment to a hard permutation         #
    #    via the Hungarian algorithm (linear_sum_assignment minimises,     #
    #    so we pass −P to maximise the probability mass).                  #
    # ------------------------------------------------------------------ #
    row_ind, col_ind = linear_sum_assignment(-P)
    # row_ind is always [0, 1, ..., n-1]; col_ind[i] is the physical
    # index (into phys_qubits) assigned to padded-logical index i.

    # ------------------------------------------------------------------ #
    # 7. Populate mapping dictionaries (1-to-1 guarantee)                 #
    # ------------------------------------------------------------------ #
    self.mapping_dict         = {}
    self.reverse_mapping_dict = {}

    for li in range(n_l):
        lq = logical_qubits[li]
        pq = phys_qubits[col_ind[li]]
        self.mapping_dict[lq]         = pq
        self.reverse_mapping_dict[pq] = lq

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

---

**How each DSSPGF stage maps to the code:**

| Stage | Code section |
|---|---|
| **Birkhoff relaxation** | `P` is an `n×n` float matrix; Sinkhorn forces row/col sums = 1 |
| **Cost proxy** | `f(P) = trace(P^T C_pad P D)` — differentiable SWAP-distance surrogate |
| **Gradient flow** | `grad = 2·C_pad·P·D`, gradient-descent step, then re-project |
| **Sinkhorn re-projection** | Alternating row/column normalisation after every gradient step |
| **Hungarian rounding** | `linear_sum_assignment(-P)` gives the nearest hard permutation |
| **1-to-1 guarantee** | Hungarian on a square matrix always returns a perfect matching |

**Computational complexity:** `O(n_steps · n³)` for the matrix multiplications plus `O(n³)` for Hungarian — fully polynomial, no exponential search.
