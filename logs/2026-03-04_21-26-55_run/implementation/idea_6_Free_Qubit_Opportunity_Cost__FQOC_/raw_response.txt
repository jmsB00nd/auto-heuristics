Looking at the FQOC idea, I need to:

1. Identify **busy** physical qubits (those occupied by front/extended layer gates)
2. For each gate pair `(Q1, Q2)`, find intermediate qubits on shortest paths using the triangle identity: `dist(Q1, Qi) + dist(Qi, Q2) == dist(Q1, Q2)`
3. Count how many of those intermediaries are **free** (not busy)
4. Scale cost as `distance / (free_intermediaries + 1)` — more free routing resources = cheaper

```python
def qlosure_poly_heuristic(self, swap_gate):
    W = 1
    front_layer_size = len(self.front_layer)
    extended_layer_size = len(self.extended_layer)

    max_decay = max(self.decay_parameter[swap_gate[0]], self.decay_parameter[swap_gate[1]])

    # Collect physical qubits actively occupied by front + extended layer gates
    busy_qubits = set()
    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        busy_qubits.add(self.temp_mapping_dict[q1])
        busy_qubits.add(self.temp_mapping_dict[q2])
    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        busy_qubits.add(self.temp_mapping_dict[q1])
        busy_qubits.add(self.temp_mapping_dict[q2])

    n_qubits = len(self.distance_matrix)
    all_qubits = range(n_qubits)

    def fqoc_score(Q1, Q2):
        d = self.distance_matrix[Q1][Q2]
        if d == 0:
            return 0.0
        if d == 1:
            # Directly adjacent: no intermediaries needed, minimal cost
            return 1.0

        # A qubit Qi lies on *some* shortest path iff the triangle equality holds
        free_intermediaries = sum(
            1
            for Qi in all_qubits
            if Qi != Q1 and Qi != Q2
            and self.distance_matrix[Q1][Qi] + self.distance_matrix[Qi][Q2] == d
            and Qi not in busy_qubits
        )

        # More free routing intermediaries → lower opportunity cost
        return d / (free_intermediaries + 1)

    f_distance = 0.0
    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        deps = self.dag_dependencies_count[g]
        f_distance += (deps + 1) * fqoc_score(Q1, Q2)

    e_distance = 0.0
    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        layer_factor = self.extended_layer_index.get(g, 0) + 1
        deps = self.dag_dependencies_count[g]
        e_distance += (deps + 1) * fqoc_score(Q1, Q2) / layer_factor

    H = max_decay * (
        f_distance / front_layer_size
        + W * ((e_distance / extended_layer_size) if extended_layer_size else 0)
    )

    return H
```

**Key differences from the baseline:**

| Aspect | Baseline | FQOC |
|---|---|---|
| Gate cost | raw `distance` | `distance / (free_intermediaries + 1)` |
| Routing context | ignored | counts free qubits on shortest paths |
| Cost signal | purely geometric | geometric + resource availability |

The `distance_matrix` triangle identity `dist(Q1,Qi) + dist(Qi,Q2) == d` is the key insight — it identifies all qubits that could serve as valid routing hops without lengthening the path. Dividing by `(free_count + 1)` encodes the core intuition: a SWAP is cheap when the path has plenty of uncontested intermediaries, and expensive when every intermediate qubit is already fighting for routing bandwidth.
