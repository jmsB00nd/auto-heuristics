Here is the implementation:

```python
def qlosure_poly_heuristic(self, swap_gate):
    a, b = swap_gate[0], swap_gate[1]

    front_layer_size = len(self.front_layer)
    extended_layer_size = len(self.extended_layer)

    max_decay = max(self.decay_parameter[a], self.decay_parameter[b])

    # --- Baseline distance cost (weighted by gate deps) ---
    f_distance = 0.0
    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        deps = self.dag_dependencies_count[g]
        f_distance += (deps + 1) * self.distance_matrix[Q1][Q2]

    e_distance = 0.0
    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        layer_factor = self.extended_layer_index.get(g, 0) + 1
        deps = self.dag_dependencies_count[g]
        e_distance += (deps + 1) * self.distance_matrix[Q1][Q2] / layer_factor

    baseline = max_decay * (
        f_distance / front_layer_size
        + ((e_distance / extended_layer_size) if extended_layer_size else 0.0)
    )

    # --- Routing Path Mutual Benefit: reward SWAPs at path intersections ---
    # Edge (a,b) lies on a shortest path Q1->Q2 iff:
    #   dist(Q1,a) + 1 + dist(b,Q2) == dist(Q1,Q2)  [forward direction]
    #   dist(Q1,b) + 1 + dist(a,Q2) == dist(Q1,Q2)  [reverse direction]
    # Urgency = (deps+1) / dist: gates needing fewer follow-up SWAPs are more urgent.

    benefit = 0.0

    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        deps = self.dag_dependencies_count[g]
        d = self.distance_matrix[Q1][Q2]
        if d == 0:
            continue
        on_path = (
            self.distance_matrix[Q1][a] + 1 + self.distance_matrix[b][Q2] == d
            or self.distance_matrix[Q1][b] + 1 + self.distance_matrix[a][Q2] == d
        )
        if on_path:
            # Urgency: high-dependency, close-to-ready gates contribute most
            urgency = (deps + 1) / d
            benefit += urgency

    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        layer_factor = self.extended_layer_index.get(g, 0) + 1
        deps = self.dag_dependencies_count[g]
        d = self.distance_matrix[Q1][Q2]
        if d == 0:
            continue
        on_path = (
            self.distance_matrix[Q1][a] + 1 + self.distance_matrix[b][Q2] == d
            or self.distance_matrix[Q1][b] + 1 + self.distance_matrix[a][Q2] == d
        )
        if on_path:
            urgency = (deps + 1) / (d * layer_factor)
            benefit += urgency

    # Normalize benefit by total gate count to keep scale comparable to baseline
    total_gates = front_layer_size + (extended_layer_size if extended_layer_size else 0)
    normalized_benefit = benefit / max(total_gates, 1)

    # Final cost: subtract benefit — SWAPs serving many gates' paths are preferred
    cost = baseline - normalized_benefit

    return cost
```

**How it differs from the baseline:**

| Aspect | Baseline | RPMBC |
|---|---|---|
| Core metric | Distance minimization | Path intersection counting |
| SWAP value | Not measured | Explicitly quantified via benefit |
| Urgency | `(deps+1) * dist` | `(deps+1) / dist` — inverse: closeness = urgency |
| Structure | Pure additive cost | Additive cost **minus** multi-gate benefit |

The critical novelty is the **path intersection test**: `dist(Q1,a) + 1 + dist(b,Q2) == dist(Q1,Q2)` checks in O(1) (using the precomputed distance matrix) whether the candidate swap edge lies on a shortest routing path. A SWAP at a high-traffic path intersection amortizes its cost across many gates simultaneously, and the benefit term captures this mathematically.
