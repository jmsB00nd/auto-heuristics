## Idea: Gate Depth-Weighted Topological Distance Cost (GDWTDC)
## Description: Normalize each gate's depth position in the circuit: depth_ratio_g = depth_g / total_depth. Multiply the distance contribution by depth_ratio_g so that early-circuit gates (small ratio, must be resolved soon) are weighted higher, and late-circuit gates contribute less. This creates a "time-urgency gradient" over the circuit.

def qlosure_poly_heuristic(self, swap_gate):
    total_depth = max(self.get_circuit_depth(), 1)  # avoid division by zero

    front_layer_size = max(len(self.front_layer), 1)
    extended_layer_size = len(self.extended_layer)

    max_decay = max(
        self.decay_parameter[swap_gate[0]],
        self.decay_parameter[swap_gate[1]]
    )

    # --- Front layer: high urgency for early gates ---
    f_cost = 0.0
    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]

        # Estimate gate's circuit-time position from physical qubit depths
        gate_depth = max(self.qubit_depth.get(Q1, 0), self.qubit_depth.get(Q2, 0))
        depth_ratio = gate_depth / total_depth          # in [0, 1]
        # Urgency: early gates (depth_ratio ≈ 0) → urgency ≈ 1 (must resolve soon)
        #          late  gates (depth_ratio ≈ 1) → urgency ≈ 0 (can wait)
        urgency = 1.0 - depth_ratio

        dist = self.distance_matrix[Q1][Q2]
        # Base distance scaled by a (1 + urgency) factor so urgency ∈ [1, 2]
        f_cost += (1.0 + urgency) * dist

    # --- Extended (lookahead) layer: also apply urgency gradient ---
    e_cost = 0.0
    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]

        gate_depth = max(self.qubit_depth.get(Q1, 0), self.qubit_depth.get(Q2, 0))
        depth_ratio = gate_depth / total_depth
        urgency = 1.0 - depth_ratio

        # Discount lookahead gates by their distance from the front
        layer_factor = self.extended_layer_index.get(g, 0) + 1
        dist = self.distance_matrix[Q1][Q2]
        e_cost += (1.0 + urgency) * dist / layer_factor

    H = max_decay * (
        f_cost / front_layer_size
        + (e_cost / extended_layer_size if extended_layer_size else 0.0)
    )

    return H
