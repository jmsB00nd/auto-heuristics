def init_mapping(self):
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    from collections import defaultdict
    import random

    # --- Step 0: Extract logical qubits and interaction weights ---
    logical_qubit_set = set()
    interaction_weight = defaultdict(float)
    for gate_id, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1.0

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)

    # Fallback for trivial circuits
    if n_lq == 0 or not interaction_weight:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            from src.utils.isl_data_loader import dict_to_isl_map
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # --- Step 1: Hungarian seed via rearrangement-inequality cost matrix ---
    weighted_degree = defaultdict(float)
    for (q1, q2), w in interaction_weight.items():
        weighted_degree[q1] += w
        weighted_degree[q2] += w

    mean_dist = {}
    for p in physical_qubits:
        finite_dists = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        mean_dist[p] = sum(finite_dists) / len(finite_dists) if finite_dists else float('inf')

    cost_matrix = np.zeros((n_lq, n_pq))
    for i, lq in enumerate(logical_qubits):
        for j, pq in enumerate(physical_qubits):
            cost_matrix[i, j] = weighted_degree.get(lq, 0) * mean_dist.get(pq, 0)

    row_ind, col_ind = linear_sum_assignment(cost_matrix)
    hungarian_assignment = {
        logical_qubits[r]: physical_qubits[c]
        for r, c in zip(row_ind, col_ind)
    }

    # --- Helper: build mapping from assignment dict ---
    def build_mapping(assignment):
        mapping = list(range(self.num_qubits))
        reverse = list(range(self.num_qubits))
        for lq, target_phys in assignment.items():
            current_phys = mapping[lq]
            if current_phys == target_phys:
                continue
            displaced_lq = reverse[target_phys]
            mapping[lq] = target_phys
            mapping[displaced_lq] = current_phys
            reverse[target_phys] = lq
            reverse[current_phys] = displaced_lq
        return mapping, reverse

    # --- Helper: compute QAP cost ---
    def qap_cost(mapping):
        cost = 0.0
        for (q1, q2), w in interaction_weight.items():
            p1, p2 = mapping[q1], mapping[q2]
            cost += w * self.distance_matrix[p1][p2]
        return cost

    # --- Helper: compute Fiedler value and vector of mapped interaction graph ---
    def compute_fiedler(mapping):
        active_physicals = set()
        for lq in logical_qubits:
            active_physicals.add(mapping[lq])

        active_list = sorted(active_physicals)
        n = len(active_list)
        if n < 2:
            return 0.0, np.ones(n) / max(n, 1), active_list

        phys_to_idx = {p: i for i, p in enumerate(active_list)}
        adj = np.zeros((n, n))

        for (q1, q2), w in interaction_weight.items():
            p1, p2 = mapping[q1], mapping[q2]
            i1, i2 = phys_to_idx[p1], phys_to_idx[p2]
            dist = self.distance_matrix[p1][p2]
            if dist > 0 and dist != float('inf'):
                edge_w = w / dist
                adj[i1, i2] += edge_w
                adj[i2, i1] += edge_w

        L = np.diag(adj.sum(axis=1)) - adj
        eigenvalues, eigenvectors = np.linalg.eigh(L)

        lambda2 = eigenvalues[1] if n > 1 else 0.0
        fiedler_vec = eigenvectors[:, 1] if n > 1 else np.zeros(n)

        return lambda2, fiedler_vec, active_list

    # --- Helper: estimate delta_lambda2 for a swap using perturbation ---
    def estimate_delta_lambda2(mapping, reverse, p1, p2, fiedler_vec, active_list, phys_to_idx):
        l1 = reverse[p1]
        l2 = reverse[p2]

        if l1 not in weighted_degree and l2 not in weighted_degree:
            return 0.0

        n = len(active_list)
        delta_L = np.zeros((n, n))

        for (q1, q2), w in interaction_weight.items():
            old_p1 = mapping[q1]
            old_p2 = mapping[q2]

            new_pq1 = old_p1
            new_pq2 = old_p2
            if q1 == l1:
                new_pq1 = p2
            elif q1 == l2:
                new_pq1 = p1
            if q2 == l1:
                new_pq2 = p2
            elif q2 == l2:
                new_pq2 = p1

            if new_pq1 == old_p1 and new_pq2 == old_p2:
                continue

            old_dist = self.distance_matrix[old_p1][old_p2]
            old_w = (w / old_dist) if (old_dist > 0 and old_dist != float('inf')) else 0.0

            new_dist = self.distance_matrix[new_pq1][new_pq2]
            new_w = (w / new_dist) if (new_dist > 0 and new_dist != float('inf')) else 0.0

            dw = new_w - old_w

            i1 = phys_to_idx.get(old_p1)
            i2 = phys_to_idx.get(old_p2)
            if i1 is None or i2 is None:
                continue

            delta_L[i1, i2] -= dw
            delta_L[i2, i1] -= dw
            delta_L[i1, i1] += dw
            delta_L[i2, i2] += dw

        return fiedler_vec @ delta_L @ fiedler_vec

    # --- Helper: steepest-ascent local search on combined objective ---
    def local_search(mapping, reverse, max_iters=150, beta=2.0):
        best_mapping = mapping[:]
        best_reverse = reverse[:]
        best_qap = qap_cost(mapping)
        best_lambda2, _, _ = compute_fiedler(mapping)
        best_obj = best_qap - beta * best_lambda2

        current_mapping = mapping[:]
        current_reverse = reverse[:]
        current_obj = best_obj

        for _ in range(max_iters):
            lambda2, fiedler_vec, active_list = compute_fiedler(current_mapping)
            if len(active_list) < 2:
                break
            phys_to_idx = {p: i for i, p in enumerate(active_list)}

            best_swap = None
            best_swap_obj = current_obj

            active_set = set(active_list)
            swap_candidates = set()
            for p in active_set:
                for neighbor in self.backend[p]:
                    swap_candidates.add((min(p, neighbor), max(p, neighbor)))

            for (p1, p2) in swap_candidates:
                delta_l2 = estimate_delta_lambda2(
                    current_mapping, current_reverse, p1, p2,
                    fiedler_vec, active_list, phys_to_idx
                )

                l1, l2 = current_reverse[p1], current_reverse[p2]
                delta_qap = 0.0
                for (q1, q2), w in interaction_weight.items():
                    old_p_q1, old_p_q2 = current_mapping[q1], current_mapping[q2]
                    new_p_q1 = old_p_q1
                    new_p_q2 = old_p_q2
                    if q1 == l1:
                        new_p_q1 = p2
                    elif q1 == l2:
                        new_p_q1 = p1
                    if q2 == l1:
                        new_p_q2 = p2
                    elif q2 == l2:
                        new_p_q2 = p1

                    if new_p_q1 == old_p_q1 and new_p_q2 == old_p_q2:
                        continue

                    old_d = self.distance_matrix[old_p_q1][old_p_q2]
                    new_d = self.distance_matrix[new_p_q1][new_p_q2]
                    delta_qap += w * (new_d - old_d)

                new_obj = current_obj + delta_qap - beta * delta_l2

                if new_obj < best_swap_obj - 1e-9:
                    best_swap_obj = new_obj
                    best_swap = (p1, p2)

            if best_swap is None:
                break

            p1, p2 = best_swap
            l1, l2 = current_reverse[p1], current_reverse[p2]
            current_mapping[l1] = p2
            current_mapping[l2] = p1
            current_reverse[p1] = l2
            current_reverse[p2] = l1
            current_obj = best_swap_obj

            if current_obj < best_obj:
                best_obj = current_obj
                best_mapping = current_mapping[:]
                best_reverse = current_reverse[:]

        return best_mapping, best_reverse, best_obj

    # --- Step 2: Multi-start with diverse seeds ---
    seeds = []

    # Seed 1: Hungarian
    m1, r1 = build_mapping(hungarian_assignment)
    seeds.append((m1, r1))

    # Seed 2: Random permutation
    available_phys = list(physical_qubits)
    random.shuffle(available_phys)
    rand_assignment = {}
    for i, lq in enumerate(logical_qubits):
        if i < len(available_phys):
            rand_assignment[lq] = available_phys[i]
    m2, r2 = build_mapping(rand_assignment)
    seeds.append((m2, r2))

    # Seed 3: Greedy centrality (high-degree logical -> most central physical)
    sorted_logical = sorted(logical_qubits, key=lambda q: weighted_degree.get(q, 0), reverse=True)
    sorted_physical = sorted(physical_qubits, key=lambda p: mean_dist.get(p, float('inf')))
    greedy_assignment = {}
    for i, lq in enumerate(sorted_logical):
        if i < len(sorted_physical):
            greedy_assignment[lq] = sorted_physical[i]
    m3, r3 = build_mapping(greedy_assignment)
    seeds.append((m3, r3))

    # Seed 4: Another random
    available_phys2 = list(physical_qubits)
    random.shuffle(available_phys2)
    rand_assignment2 = {}
    for i, lq in enumerate(logical_qubits):
        if i < len(available_phys2):
            rand_assignment2[lq] = available_phys2[i]
    m4, r4 = build_mapping(rand_assignment2)
    seeds.append((m4, r4))

    # --- Step 3: Run local search from each seed, keep best ---
    overall_best_obj = float('inf')
    overall_best_mapping = None
    overall_best_reverse = None

    for mapping, reverse in seeds:
        refined_m, refined_r, obj = local_search(mapping, reverse, max_iters=150, beta=2.0)
        if obj < overall_best_obj:
            overall_best_obj = obj
            overall_best_mapping = refined_m
            overall_best_reverse = refined_r

    self.mapping_dict = overall_best_mapping
    self.reverse_mapping_dict = overall_best_reverse

    if self.use_isl:
        from src.utils.isl_data_loader import dict_to_isl_map
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

## Algorithm Summary

The implementation follows the ACMP idea exactly:

1. **Hungarian seed (Step 1):** Builds a rearrangement-inequality cost matrix (`weighted_degree[lq] * mean_dist[pq]`) and solves with `linear_sum_assignment` to get an initial placement.

2. **Fiedler value computation (Step 2):** Constructs the mapped weighted interaction graph where edge weight = `interaction_weight / hardware_distance`, computes its Laplacian, and extracts λ₂ (algebraic connectivity) and the Fiedler vector via `np.linalg.eigh`.

3. **Steepest-ascent local search (Step 3):** For each candidate swap (p1, p2) on adjacent hardware qubits, estimates Δλ₂ using the matrix perturbation formula `Δλ₂ ≈ v^T · ΔL · v` and computes the exact ΔQAP cost. Accepts the swap that most improves the combined objective `QAP_cost - β·λ₂`.

4. **Multi-start (Step 5):** Runs from 4 diverse seeds (Hungarian, greedy centrality, 2 random) and keeps the best result.

5. **Bijection preservation:** All mutations are swap-based on the mapping/reverse_mapping lists, guaranteeing a valid 1-to-1 mapping at all times.
