def init_mapping(self):
    """
    Interference-Aware DAG Wavefront Packing.

    Process the DAG layer-by-layer (wavefront). At each wavefront, identify all 2-qubit gates
    and their interference. Pack gates into the hardware graph by treating each wavefront as a
    small matching problem, then merge wavefront solutions greedily using temporal weighting.
    """
    from collections import defaultdict, deque
    import math
    import random

    gates_list = list(self.access.items())

    # Collect logical qubits used in the circuit
    logical_qubit_set = set()
    for _, qubits in gates_list:
        for q in qubits:
            logical_qubit_set.add(q)

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            from src.utils.python_to_isl import dict_to_isl_map
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # ---------------------------------------------------------------
    # Step 1: Partition gates into DAG layers (wavefronts) via ASAP scheduling
    # ---------------------------------------------------------------
    qubit_ready = {}
    gate_layer = {}
    for gate_id, qubits in gates_list:
        layer = max((qubit_ready.get(q, 0) for q in qubits), default=0)
        gate_layer[gate_id] = layer
        for q in qubits:
            qubit_ready[q] = layer + 1

    total_depth = max(qubit_ready.values(), default=1)

    # Group gates by layer
    layers = defaultdict(list)
    for gate_id, layer in gate_layer.items():
        layers[layer].append(gate_id)
    num_layers = max(layers.keys()) + 1 if layers else 0

    # ---------------------------------------------------------------
    # Step 2: Backward pass for slack / criticality
    # ---------------------------------------------------------------
    qubit_next_ls = defaultdict(lambda: total_depth)
    gate_latest_start = {}
    for gate_id, qubits in reversed(gates_list):
        lf = min(qubit_next_ls[q] for q in qubits)
        ls = lf - 1
        gate_latest_start[gate_id] = ls
        for q in qubits:
            qubit_next_ls[q] = ls

    gate_slack = {}
    for gate_id in gate_layer:
        gate_slack[gate_id] = gate_latest_start.get(gate_id, 0) - gate_layer[gate_id]

    max_slack = max(gate_slack.values(), default=1)
    if max_slack == 0:
        max_slack = 1

    gate_criticality = {}
    for gate_id, s in gate_slack.items():
        gate_criticality[gate_id] = 1.0 + 2.0 * (1.0 - s / max_slack)

    # ---------------------------------------------------------------
    # Step 3: Compute interaction weights with temporal decay + criticality
    # ---------------------------------------------------------------
    lq_future_interactions = defaultdict(float)
    half_life = max(num_layers / 4.0, 3.0)
    for gate_id, qubits in gates_list:
        if len(qubits) == 2:
            layer = gate_layer[gate_id]
            weight = math.exp(-layer * math.log(2) / half_life) * gate_criticality[gate_id]
            for q in qubits:
                lq_future_interactions[q] += weight

    interaction_weight = defaultdict(float)
    for gate_id, qubits in gates_list:
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            layer = gate_layer[gate_id]
            w = math.exp(-layer * math.log(2) / half_life) * gate_criticality[gate_id]
            interaction_weight[key] += w

    lq_combined = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        lq_combined[q1][q2] = w
        lq_combined[q2][q1] = w

    # ---------------------------------------------------------------
    # Step 4: Layer-by-layer wavefront placement with greedy matching
    # ---------------------------------------------------------------
    lq_to_pq = {}
    pq_used = set()

    pq_adj = {pq: [nb for nb in self.backend.get(pq, []) if nb in set(physical_qubits)]
              for pq in physical_qubits}
    pq_degree = {pq: len(pq_adj[pq]) for pq in physical_qubits}

    pq_centrality = {}
    for pq in physical_qubits:
        c = sum(
            1.0 / self.distance_matrix[pq][other]
            for other in physical_qubits
            if other != pq and self.distance_matrix[pq][other] not in (0, float('inf'))
        )
        pq_centrality[pq] = c

    def find_best_neighbor(q, anchor_pq):
        """Find best physical qubit for q, preferring neighbors of anchor_pq."""
        best_pq = None
        best_cost = float('inf')

        candidates = [pq for pq in pq_adj[anchor_pq] if pq not in pq_used]
        if not candidates:
            r2 = set()
            for nb in pq_adj[anchor_pq]:
                for nb2 in pq_adj.get(nb, []):
                    if nb2 not in pq_used and nb2 != anchor_pq:
                        r2.add(nb2)
            candidates = list(r2)
        if not candidates:
            candidates = [pq for pq in physical_qubits if pq not in pq_used]

        for pq in candidates:
            cost = self.distance_matrix[anchor_pq][pq]
            for nb, w in lq_combined[q].items():
                if nb in lq_to_pq:
                    cost += w * self.distance_matrix[pq][lq_to_pq[nb]]
            cost -= 0.01 * pq_centrality[pq]
            if cost < best_cost:
                best_cost = cost
                best_pq = pq
        return best_pq

    for layer_idx in range(num_layers):
        layer_gates = layers.get(layer_idx, [])

        two_q_gates = []
        single_q_in_layer = set()
        for gate_id in layer_gates:
            qubits = self.access[gate_id]
            if len(qubits) == 2:
                two_q_gates.append((gate_id, qubits[0], qubits[1]))
            elif len(qubits) == 1:
                single_q_in_layer.add(qubits[0])

        if not two_q_gates:
            for q in single_q_in_layer:
                if q not in lq_to_pq:
                    best_pq = max(
                        (pq for pq in physical_qubits if pq not in pq_used),
                        key=lambda p: pq_centrality[p], default=None
                    )
                    if best_pq is not None:
                        lq_to_pq[q] = best_pq
                        pq_used.add(best_pq)
            continue

        # Sort by criticality (most critical first)
        two_q_gates.sort(key=lambda g: gate_criticality[g[0]], reverse=True)

        for gate_id, q1, q2 in two_q_gates:
            q1_placed = q1 in lq_to_pq
            q2_placed = q2 in lq_to_pq

            if q1_placed and q2_placed:
                continue
            elif q1_placed and not q2_placed:
                best_pq = find_best_neighbor(q2, lq_to_pq[q1])
                lq_to_pq[q2] = best_pq
                pq_used.add(best_pq)
            elif q2_placed and not q1_placed:
                best_pq = find_best_neighbor(q1, lq_to_pq[q2])
                lq_to_pq[q1] = best_pq
                pq_used.add(best_pq)
            else:
                # Neither placed - find best hardware edge
                best_edge = None
                best_cost = float('inf')

                for pq_a in physical_qubits:
                    if pq_a in pq_used:
                        continue
                    for pq_b in pq_adj[pq_a]:
                        if pq_b in pq_used or pq_b == pq_a:
                            continue

                        cost = 0.0
                        for nb, w in lq_combined[q1].items():
                            if nb in lq_to_pq and nb != q2:
                                cost += w * self.distance_matrix[pq_a][lq_to_pq[nb]]
                        for nb, w in lq_combined[q2].items():
                            if nb in lq_to_pq and nb != q1:
                                cost += w * self.distance_matrix[pq_b][lq_to_pq[nb]]
                        cost -= 0.01 * (pq_centrality[pq_a] + pq_centrality[pq_b])

                        cost_rev = 0.0
                        for nb, w in lq_combined[q1].items():
                            if nb in lq_to_pq and nb != q2:
                                cost_rev += w * self.distance_matrix[pq_b][lq_to_pq[nb]]
                        for nb, w in lq_combined[q2].items():
                            if nb in lq_to_pq and nb != q1:
                                cost_rev += w * self.distance_matrix[pq_a][lq_to_pq[nb]]
                        cost_rev -= 0.01 * (pq_centrality[pq_a] + pq_centrality[pq_b])

                        if cost_rev < cost:
                            if cost_rev < best_cost:
                                best_cost = cost_rev
                                best_edge = (pq_b, pq_a)
                        else:
                            if cost < best_cost:
                                best_cost = cost
                                best_edge = (pq_a, pq_b)

                if best_edge is not None:
                    lq_to_pq[q1] = best_edge[0]
                    lq_to_pq[q2] = best_edge[1]
                    pq_used.add(best_edge[0])
                    pq_used.add(best_edge[1])
                else:
                    for q in [q1, q2]:
                        if q not in lq_to_pq:
                            for pq in sorted(physical_qubits, key=lambda p: pq_centrality[p], reverse=True):
                                if pq not in pq_used:
                                    lq_to_pq[q] = pq
                                    pq_used.add(pq)
                                    break

        for q in single_q_in_layer:
            if q not in lq_to_pq:
                best_pq = None
                best_score = float('inf')
                for pq in physical_qubits:
                    if pq in pq_used:
                        continue
                    score = sum(w * self.distance_matrix[pq][lq_to_pq[nb]]
                                for nb, w in lq_combined[q].items() if nb in lq_to_pq)
                    score -= 0.01 * pq_centrality[pq]
                    if score < best_score:
                        best_score = score
                        best_pq = pq
                if best_pq is not None:
                    lq_to_pq[q] = best_pq
                    pq_used.add(best_pq)

    # Place remaining unplaced logical qubits
    for q in logical_qubits:
        if q not in lq_to_pq:
            best_pq = None
            best_score = float('inf')
            for pq in physical_qubits:
                if pq in pq_used:
                    continue
                score = sum(w * self.distance_matrix[pq][lq_to_pq[nb]]
                            for nb, w in lq_combined[q].items() if nb in lq_to_pq)
                score -= 0.01 * pq_centrality[pq]
                if score < best_score:
                    best_score = score
                    best_pq = pq
            if best_pq is not None:
                lq_to_pq[q] = best_pq
                pq_used.add(best_pq)

    # ---------------------------------------------------------------
    # Step 5: Build full bijective mapping
    # ---------------------------------------------------------------
    def make_mapping(lq_phys):
        md = list(range(self.num_qubits))
        for lq, pq in lq_phys.items():
            md[lq] = pq
        assigned = set(lq_phys.values())
        remaining = [pq for pq in range(self.num_qubits) if pq not in assigned]
        ri = 0
        for lq in range(self.num_qubits):
            if lq not in lq_phys:
                md[lq] = remaining[ri]
                ri += 1
        rmd = list(range(self.num_qubits))
        for lq in range(self.num_qubits):
            rmd[md[lq]] = lq
        return md, rmd

    # ---------------------------------------------------------------
    # Step 6: QAP cost and swap utilities
    # ---------------------------------------------------------------
    def qap_cost(md):
        return sum(w * self.distance_matrix[md[q1]][md[q2]]
                   for (q1, q2), w in interaction_weight.items())

    def swap_delta(p1, p2, md, rmd):
        a, b = rmd[p1], rmd[p2]
        delta = 0.0
        for nb, w in lq_combined[a].items():
            pp = md[nb]
            if pp == p2:
                continue
            delta += w * (self.distance_matrix[p2][pp] - self.distance_matrix[p1][pp])
        for nb, w in lq_combined[b].items():
            pp = md[nb]
            if pp == p1:
                continue
            delta += w * (self.distance_matrix[p1][pp] - self.distance_matrix[p2][pp])
        return delta

    def do_swap(p1, p2, md, rmd):
        a, b = rmd[p1], rmd[p2]
        md[a], md[b] = p2, p1
        rmd[p1], rmd[p2] = b, a

    lq_by_degree = sorted(logical_qubits,
                          key=lambda lq: sum(lq_combined[lq].values()), reverse=True)

    # ---------------------------------------------------------------
    # Step 7: Local search refinement
    # ---------------------------------------------------------------
    def local_search(md, rmd, max_iters=500):
        for _ in range(max_iters):
            improved = False
            for lq in lq_by_degree:
                p1 = md[lq]
                best_d, best_p2 = -1e-9, -1
                for p2 in physical_qubits:
                    if p2 == p1:
                        continue
                    d = swap_delta(p1, p2, md, rmd)
                    if d < best_d:
                        best_d, best_p2 = d, p2
                if best_p2 != -1:
                    do_swap(p1, best_p2, md, rmd)
                    improved = True
            if not improved:
                break
        return md, rmd

    best_md, best_rmd = make_mapping(lq_to_pq)
    best_md, best_rmd = local_search(best_md, best_rmd)
    best_cost = qap_cost(best_md)

    # ---------------------------------------------------------------
    # Step 8: Simulated annealing refinement
    # ---------------------------------------------------------------
    rng = random.Random(42)
    md = best_md[:]
    rmd = best_rmd[:]
    current_cost = best_cost

    active_pqs = list(set(md[lq] for lq in logical_qubits if lq_combined[lq]))
    if len(active_pqs) < 2:
        active_pqs = physical_qubits

    n_sa_iters = max(3000, n_lq * 300)
    T_start = max(current_cost * 0.05, 0.5)
    T_end = max(current_cost * 0.00005, 1e-4)
    alpha = (T_end / T_start) ** (1.0 / n_sa_iters) if n_sa_iters > 0 else 1.0
    T = T_start

    for _ in range(n_sa_iters):
        p1, p2 = rng.sample(active_pqs, 2)
        delta = swap_delta(p1, p2, md, rmd)
        if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-delta / T)):
            do_swap(p1, p2, md, rmd)
            current_cost += delta
            if current_cost < best_cost:
                best_cost = current_cost
                best_md = md[:]
                best_rmd = rmd[:]
        T *= alpha

    # Final local search
    best_md, best_rmd = local_search(best_md, best_rmd, max_iters=600)

    self.mapping_dict = best_md
    self.reverse_mapping_dict = best_rmd

    if self.use_isl:
        from src.utils.python_to_isl import dict_to_isl_map
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

**Algorithm summary:**

1. **ASAP layer partitioning** - Gates are scheduled into wavefront layers based on qubit dependencies
2. **Criticality analysis** - Forward/backward pass computes slack; low-slack gates get higher criticality weights
3. **Temporal interaction weights** - Exponential decay favors early layers; criticality boosts important gates
4. **Layer-by-layer greedy matching** - For each wavefront:
   - 2-qubit gates sorted by criticality
   - If one qubit placed: find best neighbor of its partner on the hardware graph
   - If neither placed: search all hardware edges, try both orientations, pick minimum cost considering already-placed interaction partners
   - Centrality tie-breaking ensures high-interaction qubits land centrally
5. **Bijective completion** - Unplaced logical qubits get remaining physical qubits
6. **Local search** - Iterative best-swap refinement over full physical qubit set
7. **Simulated annealing** - Metropolis exploration for escaping local minima
8. **Final local search** - Polish the SA result

**Validity guarantees:** The `pq_used` set ensures no physical qubit is assigned twice during wavefront packing, and `make_mapping` constructs a strict bijection for all `num_qubits` positions.
