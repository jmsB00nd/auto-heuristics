def init_mapping(self):
    """
    Voronoi-Seeded Cluster Expansion Placement.

    1. Select k seeds on hardware via farthest-point sampling.
    2. Build Voronoi regions (assign each physical qubit to nearest seed).
    3. Partition logical qubits into k clusters via recursive bisection on interaction graph.
    4. Compute cost matrix for cluster-to-region assignment.
    5. Hungarian assignment of clusters to regions.
    6. Within each cluster-region pair, greedy assignment.
    7. 2-opt local search across boundaries.
    """
    from collections import defaultdict, deque
    import math

    # --- Collect logical qubits and interaction weights ---
    gates_list = list(self.access.items())
    logical_qubit_set = set()
    for _, qubits in gates_list:
        for q in qubits:
            logical_qubit_set.add(q)

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # Build weighted interaction graph
    interaction = defaultdict(float)
    interaction_adj = defaultdict(lambda: defaultdict(float))
    for _, qubits in gates_list:
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction[key] += 1.0
            interaction_adj[q1][q2] += 1.0
            interaction_adj[q2][q1] += 1.0

    pq_set = set(physical_qubits)

    # --- Step 1: Select k seeds via farthest-point sampling ---
    k = max(1, min(math.ceil(math.sqrt(n_lq)), n_pq))

    # Start with highest-degree node
    pq_degree = {pq: len(self.backend.get(pq, set())) for pq in physical_qubits}
    first_seed = max(physical_qubits, key=lambda p: pq_degree[p])
    seeds = [first_seed]

    min_dist_to_seed = {}
    for pq in physical_qubits:
        d = self.distance_matrix[pq][first_seed]
        min_dist_to_seed[pq] = d if d != float('inf') else 0

    seed_set = {first_seed}
    for _ in range(1, k):
        next_seed = max(
            (pq for pq in physical_qubits if pq not in seed_set),
            key=lambda pq: min_dist_to_seed[pq],
            default=None
        )
        if next_seed is None:
            break
        seeds.append(next_seed)
        seed_set.add(next_seed)
        for pq in physical_qubits:
            d = self.distance_matrix[pq][next_seed]
            if d != float('inf') and d < min_dist_to_seed[pq]:
                min_dist_to_seed[pq] = d

    k = len(seeds)  # actual number of seeds

    # --- Step 2: Build Voronoi regions ---
    regions = defaultdict(list)  # seed_index -> list of physical qubits
    pq_to_region = {}
    for pq in physical_qubits:
        best_seed_idx = 0
        best_dist = float('inf')
        for si, seed in enumerate(seeds):
            d = self.distance_matrix[pq][seed]
            if d < best_dist:
                best_dist = d
                best_seed_idx = si
        regions[best_seed_idx].append(pq)
        pq_to_region[pq] = best_seed_idx

    # --- Step 3: Partition logical qubits into k clusters ---
    def partition_recursive(qubits, num_parts):
        if num_parts <= 1 or len(qubits) <= 1:
            return [list(qubits)]

        half_size = len(qubits) // 2
        if half_size == 0:
            return [list(qubits)]

        qubit_set = set(qubits)

        # BFS-based balanced min-cut bisection from highest-interaction qubit
        total_int = {}
        for q in qubits:
            total_int[q] = sum(
                w for q2, w in interaction_adj[q].items() if q2 in qubit_set
            )

        start_q = max(qubits, key=lambda q: total_int[q])
        visited = set()
        cluster_a = []

        queue = deque([start_q])
        visited.add(start_q)

        while queue and len(cluster_a) < half_size:
            q = queue.popleft()
            cluster_a.append(q)
            # Add neighbors sorted by interaction weight (descending)
            neighbors = []
            for q2 in qubits:
                if q2 not in visited and q2 in qubit_set:
                    w = interaction_adj[q].get(q2, 0)
                    neighbors.append((q2, w))
            neighbors.sort(key=lambda x: x[1], reverse=True)
            for q2, _ in neighbors:
                if q2 not in visited:
                    visited.add(q2)
                    queue.append(q2)

        cluster_a_set = set(cluster_a)
        cluster_b = [q for q in qubits if q not in cluster_a_set]

        # Balance
        while len(cluster_a) < half_size and cluster_b:
            cluster_a.append(cluster_b.pop())

        left_parts = max(1, num_parts // 2)
        right_parts = num_parts - left_parts

        return (partition_recursive(cluster_a, left_parts) +
                partition_recursive(cluster_b, right_parts))

    clusters = partition_recursive(logical_qubits, k)

    # Ensure exactly k clusters
    while len(clusters) < k:
        clusters.append([])
    while len(clusters) > k:
        clusters.sort(key=len)
        smallest = clusters.pop(0)
        clusters[0] = clusters[0] + smallest

    # --- Step 4: Compute cost matrix for cluster-to-region assignment ---
    region_avg_dist = {}
    for ri in range(k):
        rq = regions[ri]
        if len(rq) <= 1:
            region_avg_dist[ri] = 1.0
        else:
            total = 0.0
            count = 0
            for i_idx in range(len(rq)):
                for j_idx in range(i_idx + 1, len(rq)):
                    d = self.distance_matrix[rq[i_idx]][rq[j_idx]]
                    if d != float('inf'):
                        total += d
                        count += 1
            region_avg_dist[ri] = total / count if count > 0 else 1.0

    cost_matrix_rows = []
    for ci, cluster in enumerate(clusters):
        row = []
        cluster_set = set(cluster)
        cluster_size = len(cluster)

        # Precompute cross-cluster and internal interaction
        cross_interaction = 0.0
        internal_interaction = 0.0
        for q in cluster:
            for q2, w in interaction_adj[q].items():
                if q2 in cluster_set:
                    internal_interaction += w
                else:
                    cross_interaction += w
        internal_interaction /= 2.0  # counted twice

        for ri in range(k):
            region_size = len(regions[ri])
            # Size mismatch penalty
            size_penalty = abs(region_size - cluster_size) * 10.0

            # Internal interaction cost: denser clusters prefer tighter regions
            internal_cost = internal_interaction * region_avg_dist[ri]

            # Cross-cluster cost: clusters with many cross-edges prefer central regions
            seed_centrality = sum(
                1.0 / self.distance_matrix[seeds[ri]][s]
                for s in seeds if s != seeds[ri]
                and self.distance_matrix[seeds[ri]][s] not in (0, float('inf'))
            )
            cross_cost = cross_interaction * (1.0 / (seed_centrality + 0.01))

            # Overflow penalty
            overflow_penalty = 0.0
            if cluster_size > region_size > 0:
                overflow_penalty = 1000.0 * (cluster_size - region_size)

            cost = size_penalty + internal_cost + cross_cost + overflow_penalty
            row.append(cost)
        cost_matrix_rows.append(row)

    # --- Step 5: Hungarian assignment of clusters to regions ---
    try:
        import numpy as np
        from scipy.optimize import linear_sum_assignment
        cost_array = np.array(cost_matrix_rows, dtype=float)
        row_ind, col_ind = linear_sum_assignment(cost_array)
        cluster_to_region = {int(r): int(c) for r, c in zip(row_ind, col_ind)}
    except ImportError:
        # Greedy fallback
        cluster_to_region = {}
        used_regions = set()
        order = sorted(range(k), key=lambda ci: len(clusters[ci]), reverse=True)
        for ci in order:
            best_ri = min(
                (ri for ri in range(k) if ri not in used_regions),
                key=lambda ri: cost_matrix_rows[ci][ri],
                default=ci % k
            )
            cluster_to_region[ci] = best_ri
            used_regions.add(best_ri)

    # --- Step 6: Within each cluster-region pair, greedy assignment ---
    lq_phys = {}  # logical -> physical
    used_physical = set()

    for ci in range(k):
        ri = cluster_to_region.get(ci, ci % k)
        seed = seeds[ri]
        region_pqs = regions[ri]
        cluster_lqs = clusters[ci]

        if not cluster_lqs:
            continue

        # Sort logical qubits by total interaction (descending)
        cluster_lqs_sorted = sorted(
            cluster_lqs,
            key=lambda q: sum(interaction_adj[q].values()),
            reverse=True
        )

        available = [pq for pq in region_pqs if pq not in used_physical]

        for lq in cluster_lqs_sorted:
            if not available:
                break
            # Pick most central available physical qubit (closest to seed)
            best_pq = min(available, key=lambda pq: self.distance_matrix[pq][seed])
            lq_phys[lq] = best_pq
            used_physical.add(best_pq)
            available.remove(best_pq)

    # Handle overflow: logical qubits that couldn't fit in their assigned region
    unmapped_lq = [q for q in logical_qubits if q not in lq_phys]
    if unmapped_lq:
        remaining_pq = [pq for pq in physical_qubits if pq not in used_physical]
        for lq in unmapped_lq:
            if not remaining_pq:
                break
            neighbors_mapped = []
            for q2, w in interaction_adj[lq].items():
                if q2 in lq_phys:
                    neighbors_mapped.append((lq_phys[q2], w))

            if neighbors_mapped:
                best_pq = min(
                    remaining_pq,
                    key=lambda pq: sum(
                        w * self.distance_matrix[pq][mpq]
                        for mpq, w in neighbors_mapped
                    )
                )
            else:
                best_pq = remaining_pq[0]

            lq_phys[lq] = best_pq
            used_physical.add(best_pq)
            remaining_pq.remove(best_pq)

    # --- Step 7: 2-opt local search across boundaries ---
    lq_by_degree = sorted(
        logical_qubits,
        key=lambda q: sum(interaction_adj[q].values()),
        reverse=True
    )

    improved = True
    max_iters = 80
    iteration = 0

    while improved and iteration < max_iters:
        improved = False
        iteration += 1
        for lq1 in lq_by_degree:
            if lq1 not in lq_phys:
                continue
            pq1 = lq_phys[lq1]
            best_delta = -1e-9
            best_lq2 = None

            for lq2 in lq_by_degree:
                if lq2 >= lq1 or lq2 not in lq_phys:
                    continue
                pq2 = lq_phys[lq2]

                # Compute swap delta
                delta = 0.0
                for nb, w in interaction_adj[lq1].items():
                    if nb in lq_phys and nb != lq2:
                        pnb = lq_phys[nb]
                        delta += w * (self.distance_matrix[pq2][pnb] -
                                      self.distance_matrix[pq1][pnb])
                for nb, w in interaction_adj[lq2].items():
                    if nb in lq_phys and nb != lq1:
                        pnb = lq_phys[nb]
                        delta += w * (self.distance_matrix[pq1][pnb] -
                                      self.distance_matrix[pq2][pnb])

                if delta < best_delta:
                    best_delta = delta
                    best_lq2 = lq2

            if best_lq2 is not None:
                pq2 = lq_phys[best_lq2]
                lq_phys[lq1], lq_phys[best_lq2] = pq2, pq1
                improved = True

    # --- Build final bijective mapping lists ---
    md = list(range(self.num_qubits))
    for lq, pq in lq_phys.items():
        md[lq] = pq

    assigned_pqs = set(lq_phys.values())
    remaining_pqs = [pq for pq in range(self.num_qubits) if pq not in assigned_pqs]
    ri = 0
    for lq in range(self.num_qubits):
        if lq not in lq_phys:
            md[lq] = remaining_pqs[ri]
            ri += 1

    rmd = list(range(self.num_qubits))
    for lq in range(self.num_qubits):
        rmd[md[lq]] = lq

    self.mapping_dict = md
    self.reverse_mapping_dict = rmd

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

## Implementation Summary

The algorithm follows all 7 steps from the specification:

1. **Farthest-point sampling** (Step 1): Selects `k = ceil(sqrt(n_logical))` seed physical qubits starting from the highest-degree node, each subsequent seed maximally distant from all prior seeds.

2. **Voronoi regions** (Step 2): Each physical qubit assigned to its nearest seed, creating disjoint spatial partitions of the hardware.

3. **Recursive bisection** (Step 3): Logical qubits partitioned into k clusters via BFS-based balanced bisection on the interaction graph, keeping strongly-interacting qubits together.

4. **Cost matrix** (Step 4): Evaluates each (cluster, region) pair using size mismatch penalty, internal interaction density weighted by region compactness, cross-cluster interactions weighted by region centrality, and overflow penalties.

5. **Hungarian assignment** (Step 5): Optimal cluster-to-region matching via `scipy.optimize.linear_sum_assignment` (with greedy fallback).

6. **Greedy within-region placement** (Step 6): Highest-interaction logical qubits placed on the most central physical qubits in their assigned region. Overflow qubits placed in remaining physical locations using interaction-weighted proximity.

7. **2-opt local search** (Step 7): Iteratively swaps pairs of logical qubit assignments when the swap reduces total weighted interaction distance, up to 80 iterations or convergence.

**Validity guarantee**: Voronoi regions are disjoint; Hungarian gives 1-to-1 cluster-region matching; used-physical tracking prevents collisions; final `make_mapping` step fills all `num_qubits` slots into a valid permutation.
