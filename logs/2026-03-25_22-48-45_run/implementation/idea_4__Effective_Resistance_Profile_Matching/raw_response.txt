def init_mapping(self):
    """
    Effective Resistance Profile Matching.

    Computes effective resistance (via Laplacian pseudoinverse) between all pairs
    in both the logical interaction graph and the hardware graph. For each qubit,
    forms a sorted resistance profile vector. Matches logical to physical qubits
    by aligning profiles via Hungarian assignment, then refines with local search.
    """
    from collections import defaultdict
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    import math
    import random

    # --- Extract logical qubits from circuit ---
    logical_qubit_set = set()
    interaction_count = defaultdict(float)
    for gate_id, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_count[key] += 1.0

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    # --- Fallback for empty circuits ---
    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)
    lq_idx = {q: i for i, q in enumerate(logical_qubits)}
    pq_idx = {q: i for i, q in enumerate(physical_qubits)}

    # ========================================================================
    # Step 1-3: Build weighted Laplacian of logical interaction graph
    #           and compute effective resistances
    # ========================================================================
    L_logical = np.zeros((n_lq, n_lq))
    for (q1, q2), w in interaction_count.items():
        i, j = lq_idx[q1], lq_idx[q2]
        L_logical[i, j] -= w
        L_logical[j, i] -= w
        L_logical[i, i] += w
        L_logical[j, j] += w

    # Pseudoinverse of Laplacian
    L_logical_pinv = np.linalg.pinv(L_logical)

    # Effective resistance: R_eff(i,j) = L+[i,i] + L+[j,j] - 2*L+[i,j]
    diag_l = np.diag(L_logical_pinv)
    R_logical = diag_l[:, None] + diag_l[None, :] - 2.0 * L_logical_pinv
    np.fill_diagonal(R_logical, 0.0)
    R_logical = np.maximum(R_logical, 0.0)  # numerical stability

    # Step 4: Sorted resistance profiles for logical qubits
    logical_profiles = []
    for i in range(n_lq):
        row = R_logical[i]
        profile = np.sort(np.concatenate([row[:i], row[i+1:]]))
        logical_profiles.append(profile)

    # ========================================================================
    # Step 5: Build Laplacian of hardware graph and compute resistance profiles
    # ========================================================================
    L_hw = np.zeros((n_pq, n_pq))
    for pq in physical_qubits:
        for nb in self.backend.get(pq, []):
            if nb in pq_idx:
                i, j = pq_idx[pq], pq_idx[nb]
                if i < j:  # count each edge once
                    L_hw[i, j] -= 1.0
                    L_hw[j, i] -= 1.0
                    L_hw[i, i] += 1.0
                    L_hw[j, j] += 1.0

    L_hw_pinv = np.linalg.pinv(L_hw)
    diag_h = np.diag(L_hw_pinv)
    R_hw = diag_h[:, None] + diag_h[None, :] - 2.0 * L_hw_pinv
    np.fill_diagonal(R_hw, 0.0)
    R_hw = np.maximum(R_hw, 0.0)

    hw_profiles = []
    for i in range(n_pq):
        row = R_hw[i]
        profile = np.sort(np.concatenate([row[:i], row[i+1:]]))
        hw_profiles.append(profile)

    # ========================================================================
    # Step 6: Cost matrix via L2 distance of resistance profiles
    # ========================================================================
    profile_len_lq = n_lq - 1 if n_lq > 1 else 0
    profile_len_pq = n_pq - 1 if n_pq > 1 else 0
    max_profile_len = max(profile_len_lq, profile_len_pq)

    def pad_profile(p, target_len):
        if len(p) >= target_len:
            return p[:target_len]
        return np.concatenate([p, np.zeros(target_len - len(p))])

    cost_matrix = np.zeros((n_lq, n_pq))
    for i in range(n_lq):
        lp = pad_profile(logical_profiles[i], max_profile_len) if max_profile_len > 0 else np.array([0.0])
        for j in range(n_pq):
            hp = pad_profile(hw_profiles[j], max_profile_len) if max_profile_len > 0 else np.array([0.0])
            cost_matrix[i, j] = np.linalg.norm(lp - hp)

    # ========================================================================
    # Step 7: Hungarian assignment
    # ========================================================================
    row_ind, col_ind = linear_sum_assignment(cost_matrix)

    lq_to_pq = {}
    for r, c in zip(row_ind, col_ind):
        lq_to_pq[logical_qubits[r]] = physical_qubits[c]

    # ========================================================================
    # Step 8: Local search refinement (swap-based hill climbing)
    # ========================================================================
    def eval_cost(assignment):
        cost = 0.0
        for (q1, q2), w in interaction_count.items():
            p1 = assignment.get(q1, q1)
            p2 = assignment.get(q2, q2)
            if p1 < len(self.distance_matrix) and p2 < len(self.distance_matrix):
                cost += w * self.distance_matrix[p1][p2]
        return cost

    best_assignment = dict(lq_to_pq)
    best_cost = eval_cost(best_assignment)
    lq_list = list(lq_to_pq.keys())

    improved = True
    max_iters = 50
    iteration = 0
    while improved and iteration < max_iters:
        improved = False
        iteration += 1
        for i in range(len(lq_list)):
            for j in range(i + 1, len(lq_list)):
                a, b = lq_list[i], lq_list[j]
                best_assignment[a], best_assignment[b] = best_assignment[b], best_assignment[a]
                new_cost = eval_cost(best_assignment)
                if new_cost < best_cost - 1e-12:
                    best_cost = new_cost
                    improved = True
                else:
                    best_assignment[a], best_assignment[b] = best_assignment[b], best_assignment[a]

    # ========================================================================
    # Build full bijection mapping_dict and reverse_mapping_dict
    # ========================================================================
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_pq in best_assignment.items():
        current_pq = mapping_dict[lq]
        if current_pq == target_pq:
            continue
        displaced_lq = reverse_mapping_dict[target_pq]
        mapping_dict[lq] = target_pq
        mapping_dict[displaced_lq] = current_pq
        reverse_mapping_dict[target_pq] = lq
        reverse_mapping_dict[current_pq] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

**Implementation walkthrough:**

1. **Steps 1-3 (Logical graph effective resistance):** Builds a weighted Laplacian `L` from the circuit's 2-qubit gate interaction counts, computes its pseudoinverse `L⁺`, then derives the effective resistance matrix via `R_eff(i,j) = L⁺[i,i] + L⁺[j,j] - 2·L⁺[i,j]`.

2. **Step 4 (Logical profiles):** For each logical qubit, extracts its row from the resistance matrix, removes the self-entry, and sorts it to form a canonical "resistance profile" vector.

3. **Step 5 (Hardware graph effective resistance):** Same Laplacian pseudoinverse procedure on the hardware coupling graph (unweighted edges).

4. **Step 6 (Cost matrix):** Computes L2 distance between each logical profile and each physical profile, padding shorter profiles with zeros when `n_lq ≠ n_pq`.

5. **Step 7 (Hungarian):** `scipy.optimize.linear_sum_assignment` on the cost matrix gives a bijective assignment minimizing total profile distance.

6. **Step 8 (Local search):** Pairwise swap hill-climbing on the actual routing cost (interaction weight × hardware distance) to refine the Hungarian solution.

7. **Bijection construction:** Extends the partial assignment to a full `num_qubits`-sized permutation using the standard displacement-swap pattern used by other heuristics in the codebase.
