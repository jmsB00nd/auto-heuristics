def init_mapping(self):
    """
    Spectral Procrustes Alignment Mapping (SPAM)

    1. Build weighted interaction Laplacian for the circuit (logical qubits).
    2. Build unweighted Laplacian for the hardware topology (physical qubits).
    3. Embed both graphs into R^k via their Fiedler / higher Laplacian
       eigenvectors (skip the trivial zero eigenvalue).
    4. Center both point clouds and align them via orthogonal Procrustes
       analysis on their k×k covariance eigenbases, with a determinant
       correction to guarantee a proper rotation (det R = +1).
    5. Compute the n_L × n_P squared-Euclidean cost matrix and solve the
       assignment problem with the Hungarian algorithm (linear_sum_assignment).
    6. Commit the resulting partial assignment into a full num_qubits bijection
       using the swap-in-place pattern, so every physical qubit is covered.
    """
    import numpy as np
    from scipy.linalg import eigh
    from scipy.optimize import linear_sum_assignment
    from collections import defaultdict

    # ── Step 1: Collect logical qubits & circuit interaction weights ─────────
    logical_qubit_set = set()
    interaction_weight = defaultdict(float)

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1.0

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_L = len(logical_qubits)
    n_P = len(physical_qubits)

    # Fallback: trivial identity when no logical qubits are present
    if n_L == 0:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    lq_idx = {lq: i for i, lq in enumerate(logical_qubits)}
    pq_idx = {pq: j for j, pq in enumerate(physical_qubits)}

    # ── Step 2: Weighted Laplacian of the circuit interaction graph ──────────
    # L_C[i,i] = sum of edge weights incident on qubit i
    # L_C[i,j] = -(weight of edge i-j)
    L_C = np.zeros((n_L, n_L), dtype=float)
    for (q1, q2), w in interaction_weight.items():
        if q1 in lq_idx and q2 in lq_idx:
            i, j = lq_idx[q1], lq_idx[q2]
            L_C[i, i] += w
            L_C[j, j] += w
            L_C[i, j] -= w
            L_C[j, i] -= w

    # ── Step 3: Unweighted Laplacian of the hardware topology ────────────────
    # Process each undirected edge exactly once (j < k_nb guard)
    L_H = np.zeros((n_P, n_P), dtype=float)
    for pq in physical_qubits:
        j = pq_idx[pq]
        for nb in self.backend[pq]:
            if nb in pq_idx:
                k_nb = pq_idx[nb]
                if j < k_nb:
                    L_H[j, j] += 1.0
                    L_H[k_nb, k_nb] += 1.0
                    L_H[j, k_nb] -= 1.0
                    L_H[k_nb, j] -= 1.0

    # ── Step 4: Spectral embedding ───────────────────────────────────────────
    # Number of embedding dimensions: skip eigenvector 0 (constant / trivial)
    # Cap at 8 to stay polynomial regardless of circuit / hardware size
    k = max(1, min(n_L - 1, n_P - 1, 8))

    def spectral_embed(L, n_nodes, n_dims):
        """Return (n_nodes, n_dims) array using eigenvectors 1..n_dims of L."""
        n_vecs = min(n_dims + 1, n_nodes)
        try:
            _, vecs = eigh(L, subset_by_index=[0, n_vecs - 1])
        except Exception:
            _, vecs = np.linalg.eigh(L)
            vecs = vecs[:, :n_vecs]
        X = vecs[:, 1:n_dims + 1]           # drop trivial eigenvector
        # Pad with zeros if fewer non-trivial eigenvectors are available
        if X.shape[1] < n_dims:
            X = np.hstack([X, np.zeros((n_nodes, n_dims - X.shape[1]))])
        return X.astype(float)

    X_C = spectral_embed(L_C, n_L, k)      # (n_L, k)
    X_H = spectral_embed(L_H, n_P, k)      # (n_P, k)

    # ── Step 5: Center both point clouds ────────────────────────────────────
    X_C -= X_C.mean(axis=0)
    X_H -= X_H.mean(axis=0)

    # ── Step 6: Orthogonal Procrustes alignment ──────────────────────────────
    # Goal: find R ∈ O(k) with det R = +1 to rotate X_H so it best aligns
    # with X_C in the sense of matching their covariance principal axes.
    #
    # Since n_L ≠ n_P we cannot form X_C.T @ X_H directly; instead we
    # align the k×k covariance eigenbases:
    #
    #   cov_C = X_C.T @ X_C   EVD→  U_C
    #   cov_H = X_H.T @ X_H   EVD→  U_H
    #   M = U_C.T @ U_H   (relative orientation, k×k)
    #   SVD(M) = U s V.T
    #   R = V D U.T   with   D = diag(1,…,1, det(V U.T))  [proper rotation]
    #   X_H_aligned = X_H @ R
    cov_C = X_C.T @ X_C
    cov_H = X_H.T @ X_H

    _, U_C = eigh(cov_C)    # columns = eigenvectors, ascending eigenvalues
    _, U_H = eigh(cov_H)

    M = U_C.T @ U_H         # (k, k)
    U, _s, Vt = np.linalg.svd(M)

    # Determinant correction: ensure det(R) = +1 (rotation, not reflection)
    det_sign = np.sign(np.linalg.det(U @ Vt))
    if det_sign == 0:
        det_sign = 1.0
    D = np.diag(np.append(np.ones(k - 1), det_sign))
    R = Vt.T @ D @ U.T      # (k, k) orthogonal, det = +1

    X_H_aligned = X_H @ R   # (n_P, k)

    # ── Step 7: Hungarian assignment ─────────────────────────────────────────
    # C[i, j] = squared Euclidean distance: X_C[i] vs X_H_aligned[j]
    diff = X_C[:, np.newaxis, :] - X_H_aligned[np.newaxis, :, :]  # (n_L, n_P, k)
    cost_matrix = np.sum(diff ** 2, axis=2)                        # (n_L, n_P)

    row_ind, col_ind = linear_sum_assignment(cost_matrix)

    lq_to_phys = {
        logical_qubits[lq_i]: physical_qubits[pq_j]
        for lq_i, pq_j in zip(row_ind, col_ind)
    }

    # ── Step 8: Build strict bijection over all num_qubits indices ───────────
    # Start from the identity permutation, swap each assigned logical qubit
    # into its target physical slot, displacing whoever is there.
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

**How it works, step by step:**

| Step | What happens | Why |
|---|---|---|
| 1 | Scan `self.access` to find all logical qubits and count 2-qubit interactions | Builds the weighted interaction graph |
| 2 | Assemble `L_C` (n_L × n_L weighted Laplacian) | Encodes circuit topology |
| 3 | Assemble `L_H` (n_P × n_P unweighted Laplacian) | Encodes hardware topology |
| 4 | `eigh` on both → keep eigenvectors 1…k (skip trivial 0) | Spectral graph embedding into R^k; `k = min(n_L-1, n_P-1, 8)` keeps cost O(n²k) |
| 5 | Center both clouds | Removes translational ambiguity |
| 6 | EVD of k×k covariances → M = U_C.T @ U_H → SVD(M) → R with det correction | Procrustes rotation aligns hardware's principal axes to circuit's principal axes without needing same-size clouds |
| 7 | Cost matrix (n_L × n_P) + `linear_sum_assignment` | Hungarian algorithm guarantees globally optimal strict injective assignment in O(n_L² n_P) |
| 8 | Swap-in-place over identity permutation | Produces a valid bijection over all `num_qubits` indices with no unassigned slots |

**Complexity:** O(n² k) for the eigensolver + O(n_L² n_P) for Hungarian — fully polynomial, no exponential search.
