```python
def init_mapping(self):
    """
    Markov Blanket Locality Placement (MBLP):

    For each logical qubit, its "Markov Blanket" (MB) is the set of qubits it
    directly interacts with in the circuit, weighted by:
        interaction_frequency × temporal_dependency_weight

    where temporal_dependency_weight = 1/(rank+1) for the gate at temporal rank
    `rank` (earlier gates carry higher weight as proxies for deeper DAG ancestors,
    since the DAG is not yet built when init_mapping is called).

    Placement maximises MB-neighborhood overlap: each logical qubit is placed so
    that the physical locations of its interaction partners fall as close as
    possible to its own assigned physical qubit.

    Algorithm:
      1. Build weighted interaction graph (freq × dep_weight).
      2. Compute per-qubit MB total weight (= sum of all edge weights).
      3. Anchor: highest-MB-weight logical → most central physical (min mean BFS dist).
      4. Greedy BFS expansion driven by a max-heap over MB edge weights:
           - Pop the highest-weight unplaced qubit.
           - Score every available physical qubit by MB-overlap gain:
               overlap(p, lq) = Σ_{lq' placed, lq' ∈ MB(lq)}
                                   mb_weight(lq, lq') / (dist(p, mapping(lq')) + 1)
           - Assign lq to argmax physical (ties broken by hardware degree).
           - Push lq's unplaced MB neighbors into the heap.
      5. Isolated qubits (no 2-qubit gates) fill remaining slots by hw degree desc.
      6. Convert to strict 1-to-1 permutation via the standard in-place swap method.
    """
    import heapq
    from collections import defaultdict

    # ------------------------------------------------------------------ #
    # Step 1 – Collect logical qubits and build MB-weighted interaction   #
    #          graph: mb_weight[q1][q2] = Σ 1/(rank+1) over shared gates  #
    # ------------------------------------------------------------------ #
    logical_qubit_set = set()
    for qubits in self.access.values():
        for q in qubits:
            logical_qubit_set.add(q)

    logical_qubits  = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits:
        self.mapping_dict         = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # Temporal order: gate keys sorted ascending = circuit execution order.
    # 1/(rank+1) gives earlier gates higher weight — they have more
    # downstream dependents and are the first constraints routing must satisfy.
    sorted_gates = sorted(self.access.keys())

    mb_weight = defaultdict(lambda: defaultdict(float))

    for rank, gate in enumerate(sorted_gates):
        qubits = self.access[gate]
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            dep_w  = 1.0 / (rank + 1)
            mb_weight[q1][q2] += dep_w
            mb_weight[q2][q1] += dep_w

    # ------------------------------------------------------------------ #
    # Step 2 – Per-qubit total MB weight                                  #
    # ------------------------------------------------------------------ #
    mb_total = {q: sum(mb_weight[q].values()) for q in logical_qubits}

    # ------------------------------------------------------------------ #
    # Step 3 – Hardware centrality anchor                                 #
    # ------------------------------------------------------------------ #
    phys_degree = {p: len(self.backend[p]) for p in physical_qubits}

    def _mean_dist(p):
        vals = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        return sum(vals) / len(vals) if vals else float('inf')

    anchor_physical = min(physical_qubits, key=_mean_dist)
    anchor_logical  = max(logical_qubits,  key=lambda q: mb_total[q])

    lq_to_phys  = {anchor_logical: anchor_physical}
    placed_phys = {anchor_physical}

    # ------------------------------------------------------------------ #
    # Step 4 – Greedy MB-overlap BFS expansion                           #
    # ------------------------------------------------------------------ #
    # Max-heap entries: (-mb_edge_weight, tiebreak_counter, logical_qubit)
    heap    = []
    counter = 0
    for nb_lq, w in mb_weight[anchor_logical].items():
        heapq.heappush(heap, (-w, counter, nb_lq))
        counter += 1

    while heap:
        _, _, lq = heapq.heappop(heap)

        if lq in lq_to_phys:
            continue

        # Score each unoccupied physical qubit by MB-overlap gain.
        # overlap(p, lq) = Σ mb_weight(lq, lq') / (dist(p, phys(lq')) + 1)
        # Closer already-placed interaction partners → higher overlap score.
        best_phys  = None
        best_score = (-float('inf'), -1)   # (overlap, hw_degree) — maximise both

        for p in physical_qubits:
            if p in placed_phys:
                continue

            overlap = 0.0
            for nb_lq2, w in mb_weight[lq].items():
                if nb_lq2 in lq_to_phys:
                    nb_phys = lq_to_phys[nb_lq2]
                    d       = self.distance_matrix[p][nb_phys]
                    overlap += w / (d + 1)

            score = (overlap, phys_degree[p])
            if score > best_score:
                best_score = score
                best_phys  = p

        if best_phys is None:
            break

        lq_to_phys[lq] = best_phys
        placed_phys.add(best_phys)

        for nb_lq2, w in mb_weight[lq].items():
            if nb_lq2 not in lq_to_phys:
                heapq.heappush(heap, (-w, counter, nb_lq2))
                counter += 1

    # ------------------------------------------------------------------ #
    # Step 5 – Isolated qubits fill remaining slots by hw degree desc     #
    # ------------------------------------------------------------------ #
    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: phys_degree[p],
        reverse=True,
    )
    unplaced = [lq for lq in logical_qubits if lq not in lq_to_phys]
    for lq, phys in zip(unplaced, remaining_phys):
        lq_to_phys[lq] = phys

    # ------------------------------------------------------------------ #
    # Step 6 – Strict 1-to-1 permutation via identity base + in-place    #
    #          swap displacement                                          #
    # ------------------------------------------------------------------ #
    mapping_dict         = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq                       = reverse_mapping_dict[target_phys]
        mapping_dict[lq]                   = target_phys
        mapping_dict[displaced_lq]         = current_phys
        reverse_mapping_dict[target_phys]  = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict         = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Design rationale — what makes this genuinely different from IFWBAP:**

| Aspect | IFWBAP (previous) | MBLP (this) |
|---|---|---|
| **Edge weight** | Raw frequency count (+1 per gate) | `1/(rank+1)` — frequency × temporal dep-proxy |
| **Next-qubit selection** | Max total weight to all placed qubits (any order) | Max-heap over individual MB edge weights (highest-priority bond first) |
| **Placement cost function** | Minimize `Σ w × dist` (distance penalty) | Maximize `Σ w / (dist+1)` (overlap reward) |
| **Physical candidate scope** | Only hardware neighbors of placed qubits (with fallback) | All unoccupied physicals — no scope pruning needed because scoring is O(|P|) |
| **Tie-breaking** | `min` over sorted candidates | `max(overlap, hw_degree)` lexicographic |

**Mathematical foundation:** The overlap score `Σ w/(d+1)` is a locality potential derived from the inverse-distance kernel on the hardware graph. Maximising it is equivalent to minimising the weighted sum of routing distances the Qlosure heuristic will face in the first routing layer, while the `1/(rank+1)` dependency weighting ensures early-circuit bottlenecks (which block the most future gates) are resolved first.

**Complexity:** `O(|gates| + |L|² × |P|)` where `|L|` = logical qubits, `|P|` = physical qubits — well within polynomial bounds for realistic hardware sizes (127 physical qubits on ibm_sherbrooke).
