```python
def init_mapping(self):
    """
    Gate-Sequence Sliding Window Adjacent-Pair Greedy Seeding (GSWAPS)

    Examines the first K 2-qubit gates in circuit order (the "seeding window")
    and greedily places each interacting pair of logical qubits on an adjacent
    pair of physical qubits, prioritizing physical adjacency. This ensures the
    first K critical interactions are already "hardware-executable" before
    routing begins, eliminating early SWAP costs entirely for those gates.
    Remaining qubits fill by hardware degree.

    Complexity: O(K * P) where K is the window size and P is the number of
    physical qubits — well within polynomial bounds.
    """

    # --- Collect all logical qubits from the circuit ---
    logical_qubit_set = set()
    for qubits in self.access.values():
        for q in qubits:
            logical_qubit_set.add(q)

    # Fallback: trivial identity for empty circuits
    if not logical_qubit_set:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    # Seeding window size: scale with circuit complexity, cap at 30
    K = min(30, len(logical_qubits) * 2)

    # Extract the first K 2-qubit gates in circuit order
    seeding_window = []
    for gate_id, qubits in self.access.items():
        if len(qubits) == 2:
            seeding_window.append((qubits[0], qubits[1]))
            if len(seeding_window) >= K:
                break

    # Precompute hardware degree for scoring placement candidates
    hw_degree = {p: len(self.backend[p]) for p in physical_qubits}

    # Placement state
    lq_to_phys = {}     # logical qubit -> assigned physical qubit
    placed_phys = set() # physical qubits already occupied

    def place(lq, pq):
        lq_to_phys[lq] = pq
        placed_phys.add(pq)

    def best_adjacent_pair():
        """
        Find the unoccupied adjacent physical pair with maximum combined degree.
        High-degree pairs sit in dense hardware regions with more routing flexibility.
        """
        best = None
        best_score = -1
        for p1 in physical_qubits:
            if p1 in placed_phys:
                continue
            for p2 in self.backend[p1]:
                if p2 <= p1 or p2 in placed_phys:
                    continue
                score = hw_degree[p1] + hw_degree[p2]
                if score > best_score:
                    best_score = score
                    best = (p1, p2)
        return best

    def best_free_neighbor(pq):
        """
        Among unoccupied hardware neighbors of pq, return the one with highest
        degree (most routing options for future gates).
        """
        candidates = [nb for nb in self.backend[pq] if nb not in placed_phys]
        if not candidates:
            return None
        return max(candidates, key=lambda p: hw_degree[p])

    def closest_free(pq):
        """
        When no adjacent slot is available, fall back to the nearest unoccupied
        physical qubit by BFS hop distance.
        """
        free = [p for p in physical_qubits if p not in placed_phys]
        if not free:
            return None
        return min(
            free,
            key=lambda p: (
                self.distance_matrix[pq][p]
                if self.distance_matrix[pq][p] != float('inf')
                else 1e9
            )
        )

    # --- Phase 1: Seeding window — greedy adjacent placement ---
    for lq1, lq2 in seeding_window:
        placed1 = lq1 in lq_to_phys
        placed2 = lq2 in lq_to_phys

        if placed1 and placed2:
            # Both already assigned; adjacency cannot be forced retroactively
            continue

        if not placed1 and not placed2:
            # Neither placed: land them on the best adjacent hardware pair
            pair = best_adjacent_pair()
            if pair:
                place(lq1, pair[0])
                place(lq2, pair[1])
            # If no adjacent pair available, defer to Phase 2

        elif placed1:
            # lq1 fixed; pull lq2 right next to it
            pq1 = lq_to_phys[lq1]
            nb = best_free_neighbor(pq1)
            if nb is not None:
                place(lq2, nb)
            else:
                pq = closest_free(pq1)
                if pq is not None:
                    place(lq2, pq)

        else:  # placed2, not placed1
            # lq2 fixed; pull lq1 right next to it
            pq2 = lq_to_phys[lq2]
            nb = best_free_neighbor(pq2)
            if nb is not None:
                place(lq1, nb)
            else:
                pq = closest_free(pq2)
                if pq is not None:
                    place(lq1, pq)

    # --- Phase 2: Fill remaining logical qubits by hardware degree (desc) ---
    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: hw_degree[p],
        reverse=True
    )
    unmapped_lq = [lq for lq in logical_qubits if lq not in lq_to_phys]
    for lq, pq in zip(unmapped_lq, remaining_phys):
        place(lq, pq)

    # --- Phase 3: Build strict 1-to-1 bijection over all num_qubits indices ---
    # Start from the identity permutation; apply GSWAPS assignments via in-place
    # swaps so every physical index remains covered exactly once.
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        # Displace the logical qubit currently occupying target_phys
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Design walkthrough:**

**Phase 1 — Seeding window (the core idea)**

Iterates the first `K = min(30, 2 * num_logical)` 2-qubit gates in circuit order and handles four cases per gate:

| `lq1` placed? | `lq2` placed? | Action |
|---|---|---|
| No | No | Find the unoccupied adjacent hardware pair with the highest combined degree; place both there |
| Yes | No | Place `lq2` on the highest-degree unoccupied neighbor of `lq1`'s physical qubit; fall back to nearest-by-distance if no neighbor is free |
| No | Yes | Symmetric — pull `lq1` next to `lq2` |
| Yes | Yes | Skip (already placed; retroactive adjacency is not possible) |

The combined-degree scoring for pairs ensures the seeded qubits land in **dense hardware regions**, giving later gates in the window the best chance of also finding free adjacent slots.

**Phase 2 — Degree-based fill**

Any logical qubit not touched by the seeding window is assigned to the remaining physical qubits sorted by degree descending — placing less-constrained qubits in well-connected positions.

**Phase 3 — Identity-swap bijection**

The `lq_to_phys` partial map is applied to an identity permutation via in-place swaps (identical pattern to the other heuristics in the codebase), guaranteeing a strict 1-to-1 mapping over all `num_qubits` indices with no search overhead.
