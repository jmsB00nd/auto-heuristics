def init_mapping(self):
    from collections import defaultdict

    # --- Step 1: Compute per-gate topological layer via per-qubit sweep ---
    # Process gates in program order (dict insertion order, Python 3.7+).
    # Each gate's layer = max last-layer across its qubits, then advance
    # every qubit's cursor by 1.  O(|gates| * max_arity), no full DAG needed.
    lam = 0.5  # exponential decay rate λ for TLEDIW
    qubit_layer = defaultdict(int)  # qubit -> layer index of its last gate
    gate_layer = {}
    for gate, qubits in self.access.items():
        layer = max((qubit_layer[q] for q in qubits), default=0)
        gate_layer[gate] = layer
        for q in qubits:
            qubit_layer[q] = layer + 1

    # --- Step 2: Build time-discounted weighted interaction graph (TLEDIW) ---
    # w(gate) = exp(-λ * layer_index)  — early gates get exponentially higher
    # weight, encouraging their qubit pairs to land on adjacent physical nodes.
    interaction_weight = defaultdict(float)
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            layer = gate_layer.get(gate, 0)
            interaction_weight[key] += math.exp(-lam * layer)

    weighted_degree = defaultdict(float)
    interaction_neighbors = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        weighted_degree[q1] += w
        weighted_degree[q2] += w
        interaction_neighbors[q1][q2] = interaction_neighbors[q1].get(q2, 0) + w
        interaction_neighbors[q2][q1] = interaction_neighbors[q2].get(q1, 0) + w

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    # Fallback to trivial identity mapping if circuit has no gates
    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # --- Step 3: Anchor logical qubit = highest TLEDIW weighted degree ---
    anchor_logical = max(logical_qubits, key=lambda q: weighted_degree[q])

    # --- Step 4: Most globally-central physical qubit = lowest mean BFS distance ---
    def mean_bfs_dist(p):
        finite = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        return sum(finite) / len(finite) if finite else float('inf')

    anchor_physical = min(physical_qubits, key=mean_bfs_dist)

    # --- Step 5: Place anchor ---
    lq_to_phys = {anchor_logical: anchor_physical}
    placed_phys = {anchor_physical}
    unplaced = [lq for lq in logical_qubits if lq != anchor_logical]

    # --- Step 6: Dual BFS expansion guided by TLEDIW weights ---
    while unplaced:
        # Select next logical: maximum total TLEDIW interaction weight to placed qubits
        next_lq = max(
            unplaced,
            key=lambda lq: sum(interaction_neighbors[lq].get(placed_lq, 0)
                               for placed_lq in lq_to_phys)
        )

        # Candidate physical qubits: unoccupied hardware neighbors of placed qubits
        candidates = list({
            nb
            for phys in placed_phys
            for nb in self.backend[phys]
            if nb not in placed_phys
        })

        # Fallback: all remaining unoccupied physical qubits
        if not candidates:
            candidates = [p for p in physical_qubits if p not in placed_phys]

        if not candidates:
            break  # More logical qubits than physical — should not occur

        # Place on candidate minimizing total TLEDIW-weighted distance to placed partners
        def placement_cost(phys_c):
            total = 0.0
            for placed_lq, placed_phys_q in lq_to_phys.items():
                w = interaction_neighbors[next_lq].get(placed_lq, 0)
                if w > 0:
                    d = self.distance_matrix[phys_c][placed_phys_q]
                    total += w * (d if d != float('inf') else 1e9)
            return total

        best_phys = min(candidates, key=placement_cost)
        lq_to_phys[next_lq] = best_phys
        placed_phys.add(best_phys)
        unplaced.remove(next_lq)

    # --- Step 7: Remaining isolated qubits fill by hardware degree (descending) ---
    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: len(self.backend[p]),
        reverse=True
    )
    for lq, phys in zip(unplaced, remaining_phys):
        lq_to_phys[lq] = phys

    # --- Build strict 1-to-1 bijection over all num_qubits indices ---
    # Start from identity permutation, apply TLEDIW assignments via in-place swaps
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        # Logical qubit currently occupying target_phys must be displaced
        displaced_lq = reverse_mapping_dict[target_phys]
        # Perform swap in both directions
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

**What changed and why:**

The only structural difference from the existing IFWBAP code is in Steps 1–2. Everything from Step 3 onward is identical to the prior implementation.

**Step 1 — Layer assignment (O(G · a) where a = gate arity):**
We sweep `self.access` in program order. For each gate we take `layer = max(qubit_layer[q] for q in qubits)` — this is a data-flow lower bound on the gate's topological depth. No full DAG traversal is needed; the per-qubit cursor is sufficient because every true dependency is captured by the qubit's write-before-read ordering.

**Step 2 — Exponential time discount:**
```
interaction_weight[(q1,q2)] += exp(-λ * layer)
```
With `λ = 0.5`:
- Layer 0 → weight 1.0
- Layer 2 → weight 0.37
- Layer 5 → weight 0.08

This means qubit pairs that interact *early* accumulate far more weight than pairs that interact late. The anchor qubit (highest weighted degree) and BFS expansion then naturally cluster the most time-critical pairs onto adjacent hardware nodes, directly reducing the SWAPs needed at circuit start — the region routing cannot retroactively fix.

**Validity guarantees preserved:** The bijection construction (Steps 6–7) is unchanged — it performs in-place swaps on an identity permutation, so `mapping_dict` and `reverse_mapping_dict` remain strict 1-to-1 over `[0, num_qubits)`.
