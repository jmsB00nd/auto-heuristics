```python
def init_mapping(self):
    from collections import defaultdict
    from scipy.optimize import linear_sum_assignment
    import numpy as np

    # --- Step 1: Build interaction weight matrix W[q1][q2] ---
    # W[q][q2] = total number of 2-qubit gates between logical qubits q and q2.
    # We also collect every logical qubit, including 1-qubit-only ones.
    W = defaultdict(lambda: defaultdict(float))
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            W[q1][q2] += 1.0
            W[q2][q1] += 1.0

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    n_logical  = len(logical_qubits)
    n_physical = len(physical_qubits)

    # --- Step 2: Precompute sorted distances from each physical qubit ---
    # sorted_dists[p] = distances to all other physical qubits, ascending.
    # The k-th entry represents the best achievable distance for the k-th partner.
    sorted_dists = {}
    for p in physical_qubits:
        dists = sorted(
            self.distance_matrix[p][p2]
            for p2 in physical_qubits
            if p2 != p and self.distance_matrix[p][p2] != float('inf')
        )
        sorted_dists[p] = dists

    # --- Step 3: Build cost matrix (n_logical x n_physical) ---
    # cost[i][j] = expected SWAP overhead if logical_qubits[i] is placed at
    # physical_qubits[j].
    #
    # Estimation: sort q's interaction partners by weight descending.
    # Optimistically, the k-th heaviest partner will occupy the k-th closest
    # available physical slot near j.  Cost = sum_k  W[q][q_k] * dist_k(j).
    cost_matrix = np.zeros((n_logical, n_physical))

    for i, lq in enumerate(logical_qubits):
        # Partners sorted by interaction frequency, heaviest first
        partners = sorted(W[lq].items(), key=lambda x: -x[1])
        for j, pq in enumerate(physical_qubits):
            dists = sorted_dists[pq]
            cost = 0.0
            for k, (_partner, weight) in enumerate(partners):
                # Use k-th smallest distance; clamp to last entry if fewer slots exist
                d = dists[k] if k < len(dists) else (dists[-1] if dists else 0)
                cost += weight * d
            cost_matrix[i][j] = cost

    # --- Step 4: Solve min-cost bipartite matching (Hungarian algorithm) ---
    # scipy handles rectangular matrices (n_logical <= n_physical) natively.
    row_ind, col_ind = linear_sum_assignment(cost_matrix)

    # Build lq -> target_phys assignment from the solver output
    lq_to_phys = {
        logical_qubits[i]: physical_qubits[j]
        for i, j in zip(row_ind, col_ind)
    }

    # --- Step 5: Apply assignment via bijection-preserving swaps ---
    # Start from identity; swap entries pairwise to place each logical qubit
    # at its assigned physical qubit without breaking the bijection.
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]

        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Algorithm walkthrough:**

**Step 1 — Interaction weights.** Scan `self.access` once to build a symmetric weight matrix `W[q1][q2]` counting how many 2-qubit gates couple each logical pair. O(|gates|).

**Step 2 — Sorted distance profiles.** For every physical qubit `p`, sort its distances to all other physical qubits ascending. This gives `sorted_dists[p][k]` = "k-th nearest neighbor distance". O(P² log P).

**Step 3 — Cost matrix.** For logical qubit `lq` at candidate physical `pq`:
```
cost[lq][pq] = Σ_k  W[lq][partner_k] × sorted_dists[pq][k]
```
where partners are sorted heaviest-first. The intuition: the most-interacting partner of `lq` will ideally land at the 1st closest slot, the 2nd-heaviest at the 2nd closest, etc. This gives a lower-bound estimate of SWAP overhead without needing to know the joint assignment. O(L × P × max_degree).

**Step 4 — Hungarian algorithm.** `scipy.optimize.linear_sum_assignment` solves the min-cost bipartite matching in O(n³) — at most ~127³ ≈ 2M ops for a 127-qubit backend. Provably optimal under the local cost model.

**Step 5 — Bijection-preserving injection.** The identity list `[0,1,...,N-1]` is mutated via pairwise swaps to place each logical qubit at its assigned physical qubit, preserving the strict 1-to-1 guarantee required by the rest of the router.
