def init_mapping(self):
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    from collections import defaultdict

    # --- Step 1: Build Weighted QIG ---
    interaction_weight = defaultdict(float)
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_logical = len(logical_qubits)
    n_physical = len(physical_qubits)

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    lq_idx = {lq: i for i, lq in enumerate(logical_qubits)}
    pq_idx = {pq: i for i, pq in enumerate(physical_qubits)}

    # --- Step 2: Build QIG Laplacian (weighted by interaction counts) ---
    L_qig = np.zeros((n_logical, n_logical))
    for (q1, q2), w in interaction_weight.items():
        i, j = lq_idx.get(q1), lq_idx.get(q2)
        if i is not None and j is not None:
            L_qig[i, i] += w
            L_qig[j, j] += w
            L_qig[i, j] -= w
            L_qig[j, i] -= w

    # --- Step 3: Build Hardware Graph Laplacian (unweighted) ---
    L_hw = np.zeros((n_physical, n_physical))
    for pq in physical_qubits:
        i = pq_idx[pq]
        for nb in self.backend[pq]:
            if nb in pq_idx:
                j = pq_idx[nb]
                L_hw[i, i] += 1.0
                L_hw[i, j] -= 1.0

    # --- Step 4: Compute spectral embeddings ---
    # Use top-K non-trivial Laplacian eigenvectors as feature vectors.
    # Skip eigenvectors in the kernel (eigenvalue ≈ 0) since they carry
    # no structural information (constant on each connected component).
    K = max(1, min(5, n_logical - 1, n_physical - 1))

    def spectral_embedding(L, n, K):
        try:
            eigenvalues, eigenvectors = np.linalg.eigh(L)
        except np.linalg.LinAlgError:
            return np.zeros((n, K))
        # Select K eigenvectors with smallest strictly-positive eigenvalues
        nonzero_mask = eigenvalues > 1e-8
        nz_vecs = eigenvectors[:, nonzero_mask]
        k_actual = min(K, nz_vecs.shape[1])
        selected = nz_vecs[:, :k_actual]
        if selected.shape[1] < K:
            padding = np.zeros((n, K - selected.shape[1]))
            selected = np.hstack([selected, padding])
        return selected  # (n, K)

    embed_qig = spectral_embedding(L_qig, n_logical, K)   # (n_logical, K)
    embed_hw  = spectral_embedding(L_hw,  n_physical, K)  # (n_physical, K)

    # --- Step 5: Hungarian assignment on spectral feature space ---
    # Cost[i, j] = L2 distance between spectral vectors of logical i and physical j.
    # linear_sum_assignment finds the minimum-weight bijection in O(n^3).
    cost_matrix = np.linalg.norm(
        embed_qig[:, np.newaxis, :] - embed_hw[np.newaxis, :, :], axis=2
    )  # (n_logical, n_physical)

    row_ind, col_ind = linear_sum_assignment(cost_matrix)

    lq_to_phys = {logical_qubits[li]: physical_qubits[pi]
                  for li, pi in zip(row_ind, col_ind)}

    # Assign any leftover logical qubits (edge case: shouldn't occur on valid inputs)
    assigned_phys = set(lq_to_phys.values())
    remaining_phys = [p for p in physical_qubits if p not in assigned_phys]
    for lq, pq in zip([lq for lq in logical_qubits if lq not in lq_to_phys], remaining_phys):
        lq_to_phys[lq] = pq

    # --- Step 6: Build strict 1-to-1 bijection (swap-chain resolution) ---
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

---

**Algorithm walkthrough:**

**Step 1 — Weighted QIG:** Count each 2-qubit interaction pair as an edge weight. Single-qubit gates only contribute nodes.

**Step 2 — QIG Laplacian:** Standard normalized graph Laplacian `L = D - W` where `D[i,i]` is the weighted degree of logical qubit `i`.

**Step 3 — Hardware Laplacian:** Same construction over `self.backend` (unweighted, since topology is binary).

**Step 4 — Spectral embedding:** `np.linalg.eigh` decomposes both Laplacians. We skip all near-zero eigenvalues (they span the kernel — constant vectors with no structural content) and take the next K eigenvectors. Each qubit is now a K-dimensional point in "graph geometry" space. K is capped at 5 for speed and padded with zeros if fewer non-trivial eigenvectors exist (e.g., near-isolated nodes).

**Step 5 — Hungarian assignment:** Build an `(n_logical × n_physical)` cost matrix of Euclidean distances between spectral vectors, then call `scipy.optimize.linear_sum_assignment` which runs Jonker-Volgenant in O(n³). This guarantees a minimum-cost bijection.

**Step 6 — Bijection repair:** The swap-chain loop converts the `lq_to_phys` dict into the `mapping_dict`/`reverse_mapping_dict` lists that the rest of the system expects, maintaining the strict 1-to-1 invariant throughout.
