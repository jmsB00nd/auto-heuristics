def init_mapping(self):
    from collections import defaultdict, deque

    # ------------------------------------------------------------------ #
    # Step 1 – Build lightweight gate dependency DAG                      #
    # Gate B depends on gate A if A is the last gate to touch a qubit     #
    # that B also reads/writes (simple sequential dependency chain).      #
    # ------------------------------------------------------------------ #
    all_gates = sorted(self.access.keys())
    two_qubit_gates = [g for g in all_gates if len(self.access[g]) == 2]

    last_gate_on_qubit = {}
    successors_dag = defaultdict(set)

    for g in all_gates:
        for q in self.access[g]:
            if q in last_gate_on_qubit:
                successors_dag[last_gate_on_qubit[q]].add(g)
            last_gate_on_qubit[q] = g

    for g in all_gates:
        if g not in successors_dag:
            successors_dag[g] = set()

    # ------------------------------------------------------------------ #
    # Step 2 – Critical path length via Kahn topological sort + DP       #
    # critical_path[g] = longest path (edge count) starting at g         #
    # ------------------------------------------------------------------ #
    in_degree = defaultdict(int)
    for g in all_gates:
        if g not in in_degree:
            in_degree[g] = 0
        for s in successors_dag[g]:
            in_degree[s] += 1

    topo_order = []
    queue = deque(g for g in all_gates if in_degree[g] == 0)
    temp_in = dict(in_degree)
    while queue:
        g = queue.popleft()
        topo_order.append(g)
        for s in successors_dag[g]:
            temp_in[s] -= 1
            if temp_in[s] == 0:
                queue.append(s)

    critical_path = {g: 0 for g in all_gates}
    for g in reversed(topo_order):
        for s in successors_dag[g]:
            if critical_path[s] + 1 > critical_path[g]:
                critical_path[g] = critical_path[s] + 1

    # ------------------------------------------------------------------ #
    # Step 3 – Collect logical / physical qubits                          #
    # ------------------------------------------------------------------ #
    logical_qubit_set = set()
    for qubits in self.access.values():
        logical_qubit_set.update(qubits)

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits or not two_qubit_gates:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # ------------------------------------------------------------------ #
    # Step 4 – Transitive interaction weights between logical qubits      #
    # weight(q1,q2) += critical_path[g]+1 for every 2q gate g on (q1,q2) #
    # Gates closer to the critical root get higher weight.                #
    # ------------------------------------------------------------------ #
    interaction_neighbors = defaultdict(dict)
    for g in two_qubit_gates:
        q1, q2 = self.access[g]
        w = float(critical_path[g] + 1)
        interaction_neighbors[q1][q2] = interaction_neighbors[q1].get(q2, 0.0) + w
        interaction_neighbors[q2][q1] = interaction_neighbors[q2].get(q1, 0.0) + w

    weighted_degree = {
        q: sum(interaction_neighbors[q].values())
        for q in logical_qubits
    }

    # ------------------------------------------------------------------ #
    # Step 5 – Seed: most critical 2-qubit gate → seed qubit pair        #
    # ------------------------------------------------------------------ #
    critical_gate = max(two_qubit_gates, key=lambda g: critical_path[g])
    seed_q1, seed_q2 = self.access[critical_gate]

    # ------------------------------------------------------------------ #
    # Step 6 – Find most central ADJACENT hardware pair                   #
    # Centrality = mean BFS distance to all other reachable qubits.      #
    # Best pair (p1,p2) minimises cent[p1] + cent[p2] subject to edge.   #
    # ------------------------------------------------------------------ #
    def _centrality(p):
        finite = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        return sum(finite) / len(finite) if finite else float('inf')

    cent = {p: _centrality(p) for p in physical_qubits}

    best_phys_pair = None
    best_pair_score = float('inf')
    for p1 in physical_qubits:
        for p2 in self.backend[p1]:
            if p1 < p2:
                score = cent[p1] + cent[p2]
                if score < best_pair_score:
                    best_pair_score = score
                    best_phys_pair = (p1, p2)

    if best_phys_pair is None:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    phys_seed1, phys_seed2 = best_phys_pair

    # ------------------------------------------------------------------ #
    # Step 7 – Greedy transitive-weight fill from seed                    #
    # Each iteration: pick the unplaced lq with highest transitive        #
    # interaction weight to the already-placed set, then place it on the  #
    # BFS-frontier physical qubit that minimises weighted distance cost   #
    # (with a lookahead capacity mismatch penalty).                       #
    # ------------------------------------------------------------------ #
    lq_to_phys = {seed_q1: phys_seed1, seed_q2: phys_seed2}
    placed_phys = {phys_seed1, phys_seed2}
    unplaced = [lq for lq in logical_qubits if lq not in lq_to_phys]

    while unplaced:
        next_lq = max(
            unplaced,
            key=lambda lq: (
                sum(interaction_neighbors[lq].get(pl, 0.0) for pl in lq_to_phys),
                weighted_degree.get(lq, 0.0),
            )
        )

        candidates = list({
            nb
            for phys in placed_phys
            for nb in self.backend[phys]
            if nb not in placed_phys
        })
        if not candidates:
            candidates = [p for p in physical_qubits if p not in placed_phys]
        if not candidates:
            break

        future_neighbors = sum(
            1 for nb in interaction_neighbors[next_lq] if nb in unplaced
        )

        def _placement_score(phys_c, _lq=next_lq, _fn=future_neighbors):
            dist_cost = 0.0
            for placed_lq, placed_phys_q in lq_to_phys.items():
                w = interaction_neighbors[_lq].get(placed_lq, 0.0)
                if w > 0.0:
                    d = self.distance_matrix[phys_c][placed_phys_q]
                    dist_cost += w * (d if d != float('inf') else 1e9)
            empty_neighbors = sum(1 for nb in self.backend[phys_c] if nb not in placed_phys)
            mismatch = max(0, _fn - empty_neighbors)
            return (dist_cost, mismatch, -empty_neighbors)

        best_phys = min(candidates, key=_placement_score)
        lq_to_phys[next_lq] = best_phys
        placed_phys.add(best_phys)
        unplaced.remove(next_lq)

    # ------------------------------------------------------------------ #
    # Step 8 – Assign remaining isolated logical qubits                   #
    # ------------------------------------------------------------------ #
    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: len(self.backend[p]),
        reverse=True,
    )
    for lq, phys in zip(unplaced, remaining_phys):
        lq_to_phys[lq] = phys

    # ------------------------------------------------------------------ #
    # Step 9 – Build strict 1-to-1 bijection via in-place swap chain     #
    # ------------------------------------------------------------------ #
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

## Algorithm Walkthrough

### Step 1 — Lightweight DAG construction (O(G·Q))
For each qubit, we track the last gate that touched it. When gate `g` accesses qubit `q` and a predecessor already exists, we add `pred → g` to `successors_dag`. This gives a sparse sequential dependency DAG without requiring the full `generate_dag` machinery.

### Step 2 — Critical path DP (O(G))
Kahn's algorithm produces a topological order. Iterating in **reverse** topo order lets us compute `critical_path[g]` = length of the longest forward path from `g` in O(G + E). This is the key novelty vs plain frequency counting: gates near the root of the critical path block more downstream work.

### Step 3–4 — Transitive interaction weights
For every 2-qubit gate `g` on `(q1, q2)` we add `critical_path[g] + 1` to `interaction_neighbors[q1][q2]`. Pairs involved in early critical-path gates accumulate much larger weights than pairs that only appear deep in the DAG — precisely capturing "transitive dependency pressure."

### Step 5 — Critical gate seed
The 2-qubit gate with the **largest `critical_path` value** is the one with the most downstream transitive successors. Its qubit pair becomes the seed.

### Step 6 — Most central adjacent hardware pair
For each edge `(p1, p2)` in the hardware graph we score it by `mean_BFS_dist(p1) + mean_BFS_dist(p2)`. The most central edge minimises average future routing distance to all other qubits.

### Step 7 — Greedy BFS fill with lookahead capacity
Each round picks the unplaced logical qubit maximising **transitive interaction weight to the placed set**, then places it on the BFS-frontier physical qubit minimising a three-tuple score: `(weighted_distance_cost, capacity_mismatch_penalty, -empty_neighbors)`. The mismatch term penalises bottleneck placements where a logical qubit has more future interaction partners than the candidate physical qubit has free adjacencies.

### Guarantees
- **Validity**: the bijection step (Step 9) starts from identity and performs only displacement swaps, so the final `mapping_dict`/`reverse_mapping_dict` is always a strict permutation of `[0, num_qubits)`.
- **Complexity**: O(G log G + P²) where G = gate count, P = physical qubit count. No exponential search.
