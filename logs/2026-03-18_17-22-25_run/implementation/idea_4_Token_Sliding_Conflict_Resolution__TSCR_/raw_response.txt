```python
def init_mapping(self):
    """
    Token Sliding Conflict Resolution (TSCR) initial mapping.

    Phase 1 — Seed (IFWBAP): Place the highest-interaction-degree logical
    qubit at the most central physical qubit, then expand outward via a
    dual-BFS that always picks the unplaced logical qubit with the greatest
    total interaction weight to already-placed qubits and assigns it to the
    hardware-neighbor candidate that minimises total weighted distance to
    its interaction partners.

    Phase 2 — Token Sliding: Repeatedly find "conflict pairs" (logical qubit
    pairs that interact often but are far apart in hardware) and try sliding
    each token one step along the shortest hardware path toward its partner.
    A slide is accepted when it strictly reduces the global conflict cost
    sum_i(freq_i * dist_i).  Iteration stops when no improving slide exists.
    """
    from collections import defaultdict, deque

    # ── Phase 1: Seed placement (IFWBAP) ──────────────────────────────────

    interaction_weight = defaultdict(float)
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1

    weighted_degree = defaultdict(float)
    interaction_neighbors = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        weighted_degree[q1] += w
        weighted_degree[q2] += w
        interaction_neighbors[q1][q2] = interaction_neighbors[q1].get(q2, 0) + w
        interaction_neighbors[q2][q1] = interaction_neighbors[q2].get(q1, 0) + w

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    anchor_logical = max(logical_qubits, key=lambda q: weighted_degree[q])

    def mean_bfs_dist(p):
        finite = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        return sum(finite) / len(finite) if finite else float('inf')

    anchor_physical = min(physical_qubits, key=mean_bfs_dist)

    lq_to_phys = {anchor_logical: anchor_physical}
    placed_phys = {anchor_physical}
    unplaced = [lq for lq in logical_qubits if lq != anchor_logical]

    while unplaced:
        next_lq = max(
            unplaced,
            key=lambda lq: sum(interaction_neighbors[lq].get(placed_lq, 0)
                               for placed_lq in lq_to_phys)
        )
        candidates = list({
            nb
            for phys in placed_phys
            for nb in self.backend[phys]
            if nb not in placed_phys
        })
        if not candidates:
            candidates = [p for p in physical_qubits if p not in placed_phys]
        if not candidates:
            break

        def placement_cost(phys_c):
            total = 0.0
            for placed_lq, placed_phys_q in lq_to_phys.items():
                w = interaction_neighbors[next_lq].get(placed_lq, 0)
                if w > 0:
                    d = self.distance_matrix[phys_c][placed_phys_q]
                    total += w * (d if d != float('inf') else 1e9)
            return total

        best_phys = min(candidates, key=placement_cost)
        lq_to_phys[next_lq] = best_phys
        placed_phys.add(best_phys)
        unplaced.remove(next_lq)

    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: len(self.backend[p]),
        reverse=True
    )
    for lq, phys in zip(unplaced, remaining_phys):
        lq_to_phys[lq] = phys

    # Build the seed permutation via in-place transpositions on identity
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    # ── Phase 2: Token Sliding Conflict Resolution ─────────────────────────

    # Only pairs with at least one interaction need sliding
    conflict_pairs = [
        (q1, q2, w)
        for (q1, q2), w in interaction_weight.items()
        if w > 0
    ]

    def total_conflict_cost():
        return sum(
            w * self.distance_matrix[mapping_dict[q1]][mapping_dict[q2]]
            for q1, q2, w in conflict_pairs
        )

    def bfs_path(p_src, p_dst):
        """Shortest path on hardware graph from p_src to p_dst."""
        if p_src == p_dst:
            return [p_src]
        visited = {p_src}
        queue = deque([(p_src, [p_src])])
        while queue:
            node, path = queue.popleft()
            for nb in self.backend[node]:
                if nb == p_dst:
                    return path + [nb]
                if nb not in visited:
                    visited.add(nb)
                    queue.append((nb, path + [nb]))
        return None  # disconnected — should not occur on valid hardware

    def apply_swap(p_a, p_b):
        """Swap the logical qubit assignments of physical qubits p_a and p_b."""
        lq_a = reverse_mapping_dict[p_a]
        lq_b = reverse_mapping_dict[p_b]
        mapping_dict[lq_a] = p_b
        mapping_dict[lq_b] = p_a
        reverse_mapping_dict[p_a] = lq_b
        reverse_mapping_dict[p_b] = lq_a

    current_cost = total_conflict_cost()
    improved = True

    while improved:
        improved = False

        # Re-rank conflict pairs each pass by current severity
        conflict_pairs.sort(
            key=lambda t: -t[2] * self.distance_matrix[mapping_dict[t[0]]][mapping_dict[t[1]]]
        )

        for q1, q2, w in conflict_pairs:
            p1 = mapping_dict[q1]
            p2 = mapping_dict[q2]
            dist = self.distance_matrix[p1][p2]
            if dist <= 1:
                continue  # already adjacent — no slide needed

            path = bfs_path(p1, p2)
            if path is None or len(path) < 3:
                continue

            # Two candidate slides for this conflict pair:
            #   slide A: move q1 one step toward q2  (swap p1 ↔ path[1])
            #   slide B: move q2 one step toward q1  (swap p2 ↔ path[-2])
            for p_src, p_tgt in ((path[0], path[1]), (path[-1], path[-2])):
                apply_swap(p_src, p_tgt)
                new_cost = total_conflict_cost()

                if new_cost < current_cost:
                    current_cost = new_cost
                    improved = True
                    break  # accept slide; restart outer loop
                else:
                    apply_swap(p_src, p_tgt)  # revert (swap is its own inverse)

            if improved:
                break  # restart with updated ranking

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

## Algorithm Design

### Phase 1 — IFWBAP Seed

The seed reuses the existing placement logic verbatim, so TSCR always starts from a strong initial point rather than a random or trivial one. No extra work is duplicated.

### Phase 2 — Token Sliding Conflict Resolution

**Global cost function**
$$C = \sum_{(q_1,q_2) \in \text{2q gates}} \text{freq}(q_1,q_2) \cdot d_\text{hw}(\pi(q_1), \pi(q_2))$$

**Core loop invariant:** `mapping_dict` and `reverse_mapping_dict` always form a bijective permutation — `apply_swap` does a clean transposition of two physical slots in both directions atomically.

**Each iteration:**
1. Re-rank conflict pairs by current severity `(freq × distance)` — highest first.
2. For the top unsatisfied pair `(q1, q2)` with distance > 1:
   - Compute BFS shortest path `p1 → p2`.
   - Try **slide A**: swap `p1 ↔ path[1]` (push `q1` one hop toward `q2`).
   - Try **slide B**: swap `p2 ↔ path[-2]` (push `q2` one hop toward `q1`).
   - If either reduces `C`, accept and restart.
3. Stop when no pair yields an improving slide.

**Why `apply_swap` is its own inverse:** swapping `(p_a, p_b)` twice returns to the original state, so revert is free — no state copy needed.

**Complexity:** O(passes × pairs × |path|) with passes typically small in practice (local minima are reached quickly after a good seed).
