Here is the complete implementation:

```python
def qlosure_poly_heuristic(self, swap_gate):
    # ── Compute and cache true Ollivier-Ricci curvature distances ────────────
    if not hasattr(self, '_orcwc_distances'):
        import numpy as np
        from scipy.optimize import linprog

        n   = self.num_qubits
        INF = float('inf')

        def _wasserstein1(nbrs_u, nbrs_v):
            """
            W_1 between Uniform(nbrs_u) and Uniform(nbrs_v) using the
            graph distance matrix as the ground metric.
            Solved via the primal transportation LP.
            """
            nu, nv = len(nbrs_u), len(nbrs_v)
            if nu == 0 or nv == 0:
                return 0.0

            # Cost matrix: d(i, j) for each pair of neighbors
            c = np.array(
                [self.distance_matrix[nbrs_u[i]][nbrs_v[j]]
                 for i in range(nu) for j in range(nv)],
                dtype=float
            )

            # Supply / demand (uniform)
            supply = np.full(nu, 1.0 / nu)
            demand = np.full(nv, 1.0 / nv)

            # Equality constraints: row and column marginals
            A_eq = np.zeros((nu + nv, nu * nv))
            for i in range(nu):
                A_eq[i, i * nv:(i + 1) * nv] = 1.0
            for j in range(nv):
                A_eq[nu + j, j::nv] = 1.0

            b_eq = np.concatenate([supply, demand])
            bounds = [(0.0, None)] * (nu * nv)

            result = linprog(c, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
            return float(result.fun) if result.success else 1.0

        def _ollivier_ricci(u, v):
            """
            True Ollivier-Ricci curvature κ(u,v) = 1 - W_1(μ_u, μ_v) / d(u,v).
            μ_u = Uniform over neighbors of u.
            """
            nbrs_u = list(self.backend[u])
            nbrs_v = list(self.backend[v])
            d_uv   = self.distance_matrix[u][v]
            if d_uv == 0:
                return 1.0
            w1    = _wasserstein1(nbrs_u, nbrs_v)
            return 1.0 - w1 / d_uv

        # --- Build curvature for every hardware edge -------------------------
        curvature = {}
        for u in range(n):
            for v in self.backend[u]:
                key = (min(u, v), max(u, v))
                if key not in curvature:
                    curvature[key] = _ollivier_ricci(u, v)

        # --- Curvature-adjusted edge weights ---------------------------------
        # κ < 0  (bottleneck)  → w = 1 + α·|κ|  (expensive)
        # κ ≥ 0  (well-connected) → w = 1            (cheap)
        alpha    = 2.5                      # penalty strength for bottlenecks
        adj_dist = [[INF] * n for _ in range(n)]
        for i in range(n):
            adj_dist[i][i] = 0.0
        for u in range(n):
            for v in self.backend[u]:
                kappa           = curvature[(min(u, v), max(u, v))]
                # penalise negative curvature; leave positive curvature at 1
                w               = 1.0 + alpha * max(-kappa, 0.0)
                adj_dist[u][v]  = w

        # --- Floyd-Warshall on curvature-penalised weights -------------------
        for k in range(n):
            dk = adj_dist[k]
            for i in range(n):
                di = adj_dist[i]
                dik = di[k]
                if dik == INF:
                    continue
                for j in range(n):
                    nd = dik + dk[j]
                    if nd < di[j]:
                        di[j] = nd

        self._orcwc_distances = adj_dist

    cdist = self._orcwc_distances

    # ── Cost accumulation ────────────────────────────────────────────────────
    W                   = 0.5
    front_layer_size    = max(len(self.front_layer),    1)
    extended_layer_size = max(len(self.extended_layer), 1)

    max_decay = max(
        self.decay_parameter[swap_gate[0]],
        self.decay_parameter[swap_gate[1]]
    )

    # Front layer: Ricci-penalised routing cost, scaled by gate criticality
    f_cost = 0.0
    for g in self.front_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        if Q1 < 0 or Q2 < 0:
            continue
        deps    = self.dag_dependencies_count[g]
        f_cost += (deps + 1) * cdist[Q1][Q2]

    # Extended (lookahead) layer: same metric, attenuated by lookahead depth
    e_cost = 0.0
    for g in self.extended_layer:
        q1, q2 = self.access2q[g]
        Q1, Q2 = self.temp_mapping_dict[q1], self.temp_mapping_dict[q2]
        if Q1 < 0 or Q2 < 0:
            continue
        depth   = self.extended_layer_index.get(g, 0) + 1
        deps    = self.dag_dependencies_count[g]
        e_cost += (deps + 1) * cdist[Q1][Q2] / depth

    H = max_decay * (
        f_cost / front_layer_size
        + W * (e_cost / extended_layer_size if self.extended_layer else 0.0)
    )
    return H
```

**How it differs from ORCFC (the Forman-Ricci baseline):**

| | ORCFC (existing) | ORCWC (this) |
|---|---|---|
| Curvature formula | `2 - deg(u) - deg(v) + \|N(u)∩N(v)\|` | `1 - W₁(μ_u, μ_v) / d(u,v)` |
| Ground metric | Pure combinatorics | Graph distance matrix |
| Transport problem | None | Primal LP via `scipy.optimize.linprog` |
| Neighbor distributions | Not used | Uniform over `self.backend[node]` |

**Mathematical walkthrough:**

1. **Per-edge Ollivier-Ricci curvature** — For each hardware edge `(u,v)`, solve a small transportation LP: move the uniform mass over `u`'s neighbors to the uniform mass over `v`'s neighbors at minimum earth-mover cost `W₁`. Then `κ(u,v) = 1 - W₁/d(u,v)`.

2. **Curvature-penalised edge weight** — `w(u,v) = 1 + α·max(-κ, 0)`. Positively curved edges (rich local connectivity) cost 1. Negatively curved edges (bridge/bottleneck topology) cost up to `1 + α`.

3. **Floyd-Warshall** over `w` gives globally optimal curvature-aware paths cached on `self`.

4. **Cost aggregation** — Front and extended layers are weighted by `(deps+1)` criticality and decayed by lookahead depth, then multiplied by `max_decay`, identical in structure to the baseline but driven by an entirely different geometric distance.
