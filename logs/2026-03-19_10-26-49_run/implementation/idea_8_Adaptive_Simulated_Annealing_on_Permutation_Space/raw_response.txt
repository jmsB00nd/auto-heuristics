def init_mapping(self):
    from collections import defaultdict

    # --- Step 1: Build weighted interaction graph from circuit gates ---
    interaction_weight = defaultdict(float)
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1

    weighted_degree = defaultdict(float)
    interaction_neighbors = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        weighted_degree[q1] += w
        weighted_degree[q2] += w
        interaction_neighbors[q1][q2] = interaction_neighbors[q1].get(q2, 0) + w
        interaction_neighbors[q2][q1] = interaction_neighbors[q2].get(q1, 0) + w

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    # Fallback to trivial identity mapping if circuit has no gates
    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # --- Step 2: Frequency-greedy baseline (IFWBAP) ---
    anchor_logical = max(logical_qubits, key=lambda q: weighted_degree[q])

    def mean_bfs_dist(p):
        finite = [
            self.distance_matrix[p][o]
            for o in physical_qubits
            if o != p and self.distance_matrix[p][o] != float('inf')
        ]
        return sum(finite) / len(finite) if finite else float('inf')

    anchor_physical = min(physical_qubits, key=mean_bfs_dist)

    lq_to_phys = {anchor_logical: anchor_physical}
    placed_phys = {anchor_physical}
    unplaced = [lq for lq in logical_qubits if lq != anchor_logical]

    while unplaced:
        next_lq = max(
            unplaced,
            key=lambda lq: sum(interaction_neighbors[lq].get(placed_lq, 0)
                            for placed_lq in lq_to_phys)
        )

        candidates = list({
            nb
            for phys in placed_phys
            for nb in self.backend[phys]
            if nb not in placed_phys
        })

        if not candidates:
            candidates = [p for p in physical_qubits if p not in placed_phys]

        if not candidates:
            break

        def placement_cost(phys_c):
            total = 0.0
            for placed_lq, placed_phys_q in lq_to_phys.items():
                w = interaction_neighbors[next_lq].get(placed_lq, 0)
                if w > 0:
                    d = self.distance_matrix[phys_c][placed_phys_q]
                    total += w * (d if d != float('inf') else 1e9)
            return total

        best_phys = min(candidates, key=placement_cost)
        lq_to_phys[next_lq] = best_phys
        placed_phys.add(best_phys)
        unplaced.remove(next_lq)

    remaining_phys = sorted(
        [p for p in physical_qubits if p not in placed_phys],
        key=lambda p: len(self.backend[p]),
        reverse=True
    )
    for lq, phys in zip(unplaced, remaining_phys):
        lq_to_phys[lq] = phys

    # Build strict 1-to-1 bijection over all num_qubits indices
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq, target_phys in lq_to_phys.items():
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    # --- Step 3: Simulated Annealing refinement over permutation space ---
    # Skip SA if fewer than 2 interacting logical qubits
    if len(logical_qubits) < 2 or not interaction_weight:
        self.mapping_dict = mapping_dict
        self.reverse_mapping_dict = reverse_mapping_dict
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    INF_DIST = 1e9

    def _dist(p1, p2):
        d = self.distance_matrix[p1][p2]
        return d if d != float('inf') else INF_DIST

    def compute_cost(md):
        total = 0.0
        for (q1, q2), w in interaction_weight.items():
            total += w * _dist(md[q1], md[q2])
        return total

    def compute_delta(md, l1, l2):
        """
        Cost delta for swapping physical assignments of logical qubits l1 and l2.
        Only interaction pairs touching l1 or l2 change; the (l1, l2) pair itself
        is invariant since distance is symmetric: dist(p1, p2) == dist(p2, p1).
        Time complexity: O(deg(l1) + deg(l2)) in the interaction graph.
        """
        p1, p2 = md[l1], md[l2]
        delta = 0.0
        for lq, w in interaction_neighbors.get(l1, {}).items():
            if lq == l2:
                continue  # symmetric pair: no change
            p_lq = md[lq]
            delta += w * (_dist(p2, p_lq) - _dist(p1, p_lq))
        for lq, w in interaction_neighbors.get(l2, {}).items():
            if lq == l1:
                continue
            p_lq = md[lq]
            delta += w * (_dist(p1, p_lq) - _dist(p2, p_lq))
        return delta

    current_mapping = mapping_dict[:]
    current_rev = reverse_mapping_dict[:]
    current_cost = compute_cost(current_mapping)

    best_mapping = current_mapping[:]
    best_rev = current_rev[:]
    best_cost = current_cost

    n_lq = len(logical_qubits)

    # SA schedule: initial T proportional to cost magnitude, geometric cooling
    T = max(1.0, current_cost * 0.05)
    T_min = 1e-4
    alpha = 0.97
    steps_per_temp = max(20, n_lq * n_lq)
    max_temp_steps = 150

    rng = random.Random(42)

    for _ in range(max_temp_steps):
        if T < T_min:
            break
        for _ in range(steps_per_temp):
            idx1, idx2 = rng.sample(range(n_lq), 2)
            l1, l2 = logical_qubits[idx1], logical_qubits[idx2]

            delta = compute_delta(current_mapping, l1, l2)

            if delta < 0 or rng.random() < math.exp(-delta / T):
                # Accept: swap physical assignments of l1 and l2
                p1, p2 = current_mapping[l1], current_mapping[l2]
                current_mapping[l1] = p2
                current_mapping[l2] = p1
                current_rev[p1] = l2
                current_rev[p2] = l1
                current_cost += delta

                if current_cost < best_cost:
                    best_cost = current_cost
                    best_mapping = current_mapping[:]
                    best_rev = current_rev[:]

        T *= alpha

    self.mapping_dict = best_mapping
    self.reverse_mapping_dict = best_rev

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

## Design decisions explained

**Baseline initialization (Step 2)**  
The existing IFWBAP greedy BFS is kept exactly — it gives SA a warm, near-optimal start rather than a random permutation, drastically reducing the search space SA needs to cover.

**SA neighborhood: logical-qubit swap**  
Each move picks two logical qubits `l1, l2` and swaps their physical assignments. This directly manipulates the permutation space the problem lives in. The 1-to-1 bijection invariant is preserved automatically — swapping two entries in a permutation always yields another valid permutation.

**Incremental delta computation (O(deg) per step)**  
The cost is `Σ w(q1,q2) · dist(map[q1], map[q2])`. When swapping `l1 ↔ l2`, only pairs that include `l1` or `l2` change. The `(l1, l2)` pair itself contributes zero delta because `dist(p1,p2) = dist(p2,p1)`. So `compute_delta` runs in O(degree of l1 + degree of l2) — much cheaper than recomputing the full cost from scratch.

**SA parameters**  
| Parameter | Value | Rationale |
|---|---|---|
| `T_init` | `max(1, cost × 0.05)` | Scales with problem size; allows ~5% cost worsening at start |
| `alpha` | `0.97` | Moderate cooling; 150 steps gives ~≈ 10× temperature reduction |
| `steps_per_temp` | `n²` | Covers O(n²) possible swap pairs per temperature level |
| `max_temp_steps` | `150` | Empirically balanced; total work = 150 × n² SA steps |
| `rng = Random(42)` | fixed seed | Reproducible results |

**Best-ever tracking**  
SA stores the globally best solution seen throughout the entire run (not just the final state), which is standard for SA on combinatorial problems — the final low-temperature state may be a local minimum inferior to an earlier high-temperature discovery.
