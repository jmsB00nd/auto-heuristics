def init_mapping(self):
    from collections import defaultdict

    # --- Build weighted interaction graph ---
    interaction_weight = defaultdict(float)
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += 1

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())
    n_logical = len(logical_qubits)

    # Fallback to trivial identity mapping if circuit has no 2-qubit interactions
    if not logical_qubits or not interaction_weight:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    # Map logical qubits to dense indices 0..n_logical-1
    lq_index = {lq: i for i, lq in enumerate(logical_qubits)}

    # Build per-node adjacency list for O(degree) incremental energy updates
    # adj[i] = list of (j, w) for all interaction edges touching index i
    adj = [[] for _ in range(n_logical)]
    for (q1, q2), w in interaction_weight.items():
        i, j = lq_index[q1], lq_index[q2]
        adj[i].append((j, w))
        adj[j].append((i, w))

    dist = self.distance_matrix  # 2D list: dist[p1][p2]

    def compute_energy(perm):
        """E(π) = Σ_{(i,j)} w_ij * dist[π(i)][π(j)]"""
        E = 0.0
        for i in range(n_logical):
            pi = perm[i]
            for j, w in adj[i]:
                if j > i:  # count each edge once
                    E += w * dist[pi][perm[j]]
        return E

    def delta_energy(perm, a, b):
        """Incremental ΔE for swapping positions a and b in perm (O(deg_a + deg_b))."""
        pa, pb = perm[a], perm[b]
        dE = 0.0
        for j, w in adj[a]:
            if j == b:
                continue  # symmetric distance: this edge cost is unchanged
            pj = perm[j]
            dE += w * (dist[pb][pj] - dist[pa][pj])
        for j, w in adj[b]:
            if j == a:
                continue
            pj = perm[j]
            dE += w * (dist[pa][pj] - dist[pb][pj])
        return dE

    def run_chain(init_perm, temperature, n_steps):
        """
        Fixed-temperature Metropolis-Hastings chain.
        Returns (best_perm, best_energy) seen across all visited states.
        """
        perm = init_perm[:]
        E = compute_energy(perm)
        best_perm = perm[:]
        best_E = E
        for _ in range(n_steps):
            a = random.randrange(n_logical)
            b = random.randrange(n_logical - 1)
            if b >= a:
                b += 1  # ensure a != b without rejection sampling
            dE = delta_energy(perm, a, b)
            if dE <= 0 or (temperature > 0 and random.random() < math.exp(-dE / temperature)):
                perm[a], perm[b] = perm[b], perm[a]
                E += dE
                if E < best_E:
                    best_E = E
                    best_perm = perm[:]
        return best_perm, best_E

    # --- Temperature schedule ---
    # Estimate energy scale as average interaction weight × average hardware distance
    avg_w = sum(interaction_weight.values()) / len(interaction_weight)
    all_finite = [
        dist[p1][p2]
        for p1 in physical_qubits
        for p2 in physical_qubits
        if p1 != p2 and dist[p1][p2] != float('inf')
    ]
    avg_d = sum(all_finite) / len(all_finite) if all_finite else 2.0
    T_high = max(1.0, avg_w * avg_d * n_logical * 0.2)
    T_low  = max(0.05, T_high * 0.005)

    n_chains = 8
    # Logarithmically-spaced temperatures from T_low to T_high
    log_low, log_high = math.log(T_low), math.log(T_high)
    temperatures = [
        math.exp(log_low + (log_high - log_low) * i / max(1, n_chains - 1))
        for i in range(n_chains)
    ]

    n_steps = max(1000, n_logical * 300)

    best_overall_perm = None
    best_overall_E = float('inf')

    for T in temperatures:
        # Each chain starts from a fresh random assignment of physical qubits
        init_phys = random.sample(physical_qubits, n_logical)
        perm, E = run_chain(init_phys, T, n_steps)
        if E < best_overall_E:
            best_overall_E = E
            best_overall_perm = perm[:]

    # --- Build strict 1-to-1 bijection over all num_qubits indices ---
    # Start from identity permutation and apply MCMC assignments via in-place swaps
    mapping_dict = list(range(self.num_qubits))
    reverse_mapping_dict = list(range(self.num_qubits))

    for lq_i, target_phys in enumerate(best_overall_perm):
        lq = logical_qubits[lq_i]
        current_phys = mapping_dict[lq]
        if current_phys == target_phys:
            continue
        displaced_lq = reverse_mapping_dict[target_phys]
        mapping_dict[lq] = target_phys
        mapping_dict[displaced_lq] = current_phys
        reverse_mapping_dict[target_phys] = lq
        reverse_mapping_dict[current_phys] = displaced_lq

    self.mapping_dict = mapping_dict
    self.reverse_mapping_dict = reverse_mapping_dict

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

Here's how the algorithm works, piece by piece:

### Energy function
```
E(π) = Σ_{(q1,q2)} w(q1,q2) · dist[π(q1)][π(q2)]
```
`w(q1,q2)` = number of 2-qubit gates between logical qubits q1 and q2. This directly approximates SWAP overhead — high-frequency pairs placed far apart pay a large penalty.

### Incremental `delta_energy` (the key optimization)
Rather than recomputing the full energy on every proposal (O(edges)), `delta_energy` only touches the edges adjacent to the two swapped positions. Cost is **O(deg_a + deg_b)** per step instead of O(|E|), which is critical for dense circuits.

The math: when positions `a` and `b` swap physical assignments `pa ↔ pb`:
- Edges `(a,j)` for `j≠b`: cost changes from `w·dist[pa][pj]` → `w·dist[pb][pj]`
- Edges `(b,j)` for `j≠a`: cost changes from `w·dist[pb][pj]` → `w·dist[pa][pj]`  
- Edge `(a,b)` itself: distance is symmetric, so cost is **unchanged** (correctly skipped)

### Multi-chain strategy
8 chains run at logarithmically-spaced temperatures from `T_low ≈ 0.005·T_high` to `T_high ∝ avg_weight × avg_dist × n_logical`. This covers the full exploration-exploitation spectrum simultaneously. Every chain tracks its **personal best** (not just current state), so even a hot chain that wanders contributes its best-found permutation.

### Validity guarantee
The final bijection is built using the same swap-based identity-permutation technique used elsewhere in the codebase — every logical qubit gets exactly one physical qubit, and every physical qubit is assigned to exactly one logical qubit (or itself for unmapped indices).
