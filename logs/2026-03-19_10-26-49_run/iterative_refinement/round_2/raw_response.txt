## Analysis of Current Best Heuristic

**Strengths:**
- Rearrangement-inequality cost matrix captures both weight distribution AND local topology simultaneously
- Hungarian algorithm gives globally optimal assignment under that proxy
- 2-opt QAP refinement with efficient delta computation

**Key Weaknesses:**
1. **Interaction weights are time-blind**: all two-qubit gates get weight `+1` regardless of when they appear in the circuit. A gate at layer 50 is treated identically to one at layer 0, but only early gates create *immediate* routing pressure.
2. **Single local optimum**: the 2-opt converges to one local minimum with no escape mechanism.
3. **No topological ordering exploitation**: the circuit DAG structure is entirely ignored during cost estimation.

## Evolution Strategy: OPTION A (Targeted Refinement)

Two targeted changes:
1. **Layer-decay weighting** (`GAMMA=0.85`): weight each gate's interaction by `GAMMA^layer`. This makes the cost matrix reflect *immediate* routing pressure rather than uniform circuit statistics. Gates at the critical head of the DAG get highest weight.
2. **Iterated Local Search (ILS)**: after 2-opt convergence, apply a random 4-cycle perturbation (rotate 4 qubits through their positions) and re-optimize, keeping the global best. This escapes local optima without restarting from scratch.

```python
def init_mapping(self):
    from collections import defaultdict
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    import random

    GAMMA = 0.85  # layer-decay: earlier layers get higher weight

    # --- Step 1: Compute gate layers via topological scheduling ---
    # Assign each gate to layer = max(depth of its qubits), then advance qubit depths.
    qubit_depth = defaultdict(int)
    gate_layer = {}
    logical_qubit_set = set()

    for gate, qubits in self.access.items():
        for q in qubits:
            logical_qubit_set.add(q)
        layer = max((qubit_depth[q] for q in qubits), default=0)
        gate_layer[gate] = layer
        for q in qubits:
            qubit_depth[q] = layer + 1

    # --- Step 2: Build decay-weighted interaction graph ---
    # w(pair) += GAMMA^layer so earlier interactions dominate the cost signal.
    interaction_weight = defaultdict(float)
    for gate, qubits in self.access.items():
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            interaction_weight[key] += GAMMA ** gate_layer[gate]

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)

    # --- Step 3: Rearrangement-inequality cost matrix ---
    # For logical qubit lq at physical qubit pq:
    #   cost = sum_k  w_k * d_k
    # where w_1 >= w_2 >= ... are lq's sorted (decay-weighted) interaction weights
    # and   d_1 <= d_2 <= ... are pq's sorted distances to other physical qubits.
    lq_sorted_weights = {}
    for lq in logical_qubits:
        weights = [w for (q1, q2), w in interaction_weight.items() if q1 == lq or q2 == lq]
        lq_sorted_weights[lq] = sorted(weights, reverse=True)

    pq_sorted_dists = {}
    for pq in physical_qubits:
        dists = sorted([
            self.distance_matrix[pq][other]
            for other in physical_qubits
            if other != pq and self.distance_matrix[pq][other] != float('inf')
        ])
        pq_sorted_dists[pq] = dists

    cost_matrix = np.zeros((n_lq, n_pq))
    for i, lq in enumerate(logical_qubits):
        weights = lq_sorted_weights[lq]
        if not weights:
            continue
        for j, pq in enumerate(physical_qubits):
            dists = pq_sorted_dists[pq]
            cost = 0.0
            for k, w in enumerate(weights):
                d = dists[k] if k < len(dists) else (dists[-1] + k - len(dists) + 1)
                cost += w * d
            cost_matrix[i, j] = cost

    # --- Step 4: Hungarian assignment ---
    row_ind, col_ind = linear_sum_assignment(cost_matrix)
    lq_to_phys = {
        logical_qubits[r]: physical_qubits[c]
        for r, c in zip(row_ind, col_ind)
    }

    # --- Step 5: Build initial mapping via in-place swaps ---
    def build_mapping(lq_phys_map):
        md = list(range(self.num_qubits))
        rmd = list(range(self.num_qubits))
        for lq, target in lq_phys_map.items():
            cur = md[lq]
            if cur == target:
                continue
            disp = rmd[target]
            md[lq] = target
            md[disp] = cur
            rmd[target] = lq
            rmd[cur] = disp
        return md, rmd

    mapping_dict, reverse_mapping_dict = build_mapping(lq_to_phys)

    # --- Step 6: Interaction structures for QAP local search ---
    lq_interactions = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        lq_interactions[q1][q2] = w
        lq_interactions[q2][q1] = w

    def qap_cost(md):
        return sum(
            w * self.distance_matrix[md[q1]][md[q2]]
            for (q1, q2), w in interaction_weight.items()
        )

    def swap_delta(md, rmd, pos1, pos2):
        lq_a = rmd[pos1]
        lq_b = rmd[pos2]
        delta = 0.0
        for partner, w in lq_interactions[lq_a].items():
            pp = md[partner]
            if pp != pos2:
                delta += w * (self.distance_matrix[pos2][pp] - self.distance_matrix[pos1][pp])
        for partner, w in lq_interactions[lq_b].items():
            pp = md[partner]
            if pp != pos1:
                delta += w * (self.distance_matrix[pos1][pp] - self.distance_matrix[pos2][pp])
        return delta

    def local_search_2opt(md, rmd, max_iters=200):
        md, rmd = md[:], rmd[:]
        improved = True
        itr = 0
        while improved and itr < max_iters:
            improved = False
            itr += 1
            for lq1 in logical_qubits:
                p1 = md[lq1]
                best_delta = -1e-9
                best_p2 = -1
                for p2 in physical_qubits:
                    if p2 == p1:
                        continue
                    d = swap_delta(md, rmd, p1, p2)
                    if d < best_delta:
                        best_delta = d
                        best_p2 = p2
                if best_p2 != -1:
                    lq2 = rmd[best_p2]
                    md[lq1] = best_p2
                    md[lq2] = p1
                    rmd[p1] = lq2
                    rmd[best_p2] = lq1
                    improved = True
        return md, rmd

    # --- Step 7: Initial 2-opt refinement ---
    mapping_dict, reverse_mapping_dict = local_search_2opt(mapping_dict, reverse_mapping_dict)
    best_cost = qap_cost(mapping_dict)
    best_md = mapping_dict[:]
    best_rmd = reverse_mapping_dict[:]

    # --- Step 8: Iterated Local Search with 4-cycle perturbation ---
    # After convergence, rotate 4 qubits through their physical positions (a 4-cycle),
    # then re-run 2-opt. This is a perturbation strong enough to escape local optima
    # but weak enough to preserve most of the good structure.
    rng = random.Random(42)
    lq_list = list(logical_qubits)

    for _ in range(8):
        md, rmd = best_md[:], best_rmd[:]

        k = min(4, len(lq_list))
        if k < 2:
            break

        # Pick k logical qubits and rotate their physical positions in a cycle:
        #   chosen[0] -> pos[1], chosen[1] -> pos[2], ..., chosen[k-1] -> pos[0]
        chosen = rng.sample(lq_list, k)
        old_pos = [md[lq] for lq in chosen]
        new_pos = old_pos[1:] + [old_pos[0]]

        for lq, np_ in zip(chosen, new_pos):
            md[lq] = np_
            rmd[np_] = lq

        md, rmd = local_search_2opt(md, rmd, max_iters=150)
        cost = qap_cost(md)

        if cost < best_cost:
            best_cost = cost
            best_md = md[:]
            best_rmd = rmd[:]

    self.mapping_dict = best_md
    self.reverse_mapping_dict = best_rmd

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Summary of changes vs current best (335.33 swaps):**

| Change | Rationale |
|--------|-----------|
| `GAMMA^layer` decay weights | Gates at layer 0 get weight 1.0, layer 10 gets ~0.20 — this biases the cost matrix toward reducing immediate swap pressure at circuit start |
| Gate layer computation | Topological scheduling assigns each gate its earliest schedulable layer; no overhead since it's O(n_gates) |
| ILS with 4-cycle kick | After 2-opt converges, a k=4 rotation permutes 4 qubits' positions — strong enough to escape local optima, preserves ~n_lq−4 assignments |
| 8 ILS restarts | Each restart re-runs 2-opt from the perturbed state; global best is kept |
