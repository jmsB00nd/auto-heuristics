**Critical Analysis of Round 6:**

**Strengths:** CPM forward/backward pass, exponential-decay interaction weights, Hungarian + multi-start BFS seeding, two-tier local search (r2 + full), ILS with 4 strategies, SA with geometric cooling.

**Weaknesses/Opportunities:**
1. **Fixed ILS strategy rotation** — no feedback on which strategies work; some may waste most of the budget
2. **Static qubit ordering in local search** — `lq_by_degree` never adapts; high-cost qubits should be prioritized each iteration
3. **No crossover** between the multiple good seeds found in Step 10 — the population is discarded after seeding
4. **SA stagnation** — no reheat when acceptance rate collapses; single-phase geometric cooling wastes budget when stuck
5. **5th perturbation missing** — chain reversal (2-opt-style along the interaction graph) is a natural addition

**Evolution Strategy: OPTION A (Refine)** — targeted improvements:
- Add **crossover** between top solutions from multi-start phase
- **UCB1-adaptive ILS** strategy selection
- **Dynamic cost-guided ordering** in local search (sort qubits by their current cost contribution per iteration)
- **5th perturbation**: chain reversal along interaction graph
- **SA with periodic reheat** when acceptance rate collapses

```python
def init_mapping(self):
    from collections import defaultdict
    import numpy as np
    from scipy.optimize import linear_sum_assignment
    import math
    import random

    gates_list = list(self.access.items())

    logical_qubit_set = set()
    for _, qubits in gates_list:
        for q in qubits:
            logical_qubit_set.add(q)

    # --- Step 1: CPM forward pass ---
    qubit_ready_fwd = {}
    gate_earliest_start = []
    gate_earliest_finish = []
    for _, qubits in gates_list:
        es = max((qubit_ready_fwd.get(q, 0) for q in qubits), default=0)
        ef = es + 1
        gate_earliest_start.append(es)
        gate_earliest_finish.append(ef)
        for q in qubits:
            qubit_ready_fwd[q] = ef

    total_depth = max(gate_earliest_finish, default=1)
    max_layer = total_depth - 1

    # --- Step 2: CPM backward pass ---
    qubit_next_ls = defaultdict(lambda: total_depth)
    gate_latest_start = [0] * len(gates_list)
    for idx in range(len(gates_list) - 1, -1, -1):
        _, qubits = gates_list[idx]
        lf = min(qubit_next_ls[q] for q in qubits)
        ls = lf - 1
        gate_latest_start[idx] = ls
        for q in qubits:
            qubit_next_ls[q] = ls

    gate_slack = [gate_latest_start[i] - gate_earliest_start[i]
                  for i in range(len(gates_list))]
    max_slack = max(gate_slack, default=1) if gate_slack else 1

    if max_slack == 0:
        criticality = [2.5] * len(gates_list)
    else:
        criticality = [1.0 + 2.5 * (1.0 - s / max_slack) for s in gate_slack]

    # --- Step 3: Layer-decayed + critical-window interaction weights ---
    interaction_weight = defaultdict(float)
    half_life = max(max_layer / 3.5, 5.0)
    critical_end = max(1, int(max_layer * 0.15))
    crit_half_life = max(critical_end / 2.0, 1.0)

    for idx, (_, qubits) in enumerate(gates_list):
        if len(qubits) == 2:
            q1, q2 = qubits[0], qubits[1]
            key = (min(q1, q2), max(q1, q2))
            layer = gate_earliest_start[idx]
            w = math.exp(-layer * math.log(2) / half_life) + 0.08
            if layer <= critical_end:
                w += 1.5 * math.exp(-layer * math.log(2) / crit_half_life)
            w *= criticality[idx]
            interaction_weight[key] += w

    logical_qubits = sorted(logical_qubit_set)
    physical_qubits = sorted(self.backend.keys())

    if not logical_qubits:
        self.mapping_dict = list(range(self.num_qubits))
        self.reverse_mapping_dict = list(range(self.num_qubits))
        if self.use_isl:
            self.isl_mapping = dict_to_isl_map(self.mapping_dict)
        return

    n_lq = len(logical_qubits)
    n_pq = len(physical_qubits)
    pq_set = set(physical_qubits)

    lq_interactions = defaultdict(dict)
    for (q1, q2), w in interaction_weight.items():
        lq_interactions[q1][q2] = w
        lq_interactions[q2][q1] = w

    pq_adj = {pq: [nb for nb in self.backend.get(pq, []) if nb in pq_set]
              for pq in physical_qubits}
    pq_degree = {pq: len(pq_adj[pq]) for pq in physical_qubits}

    # --- Step 4: Physical topology analysis ---
    pq_centrality = {}
    for pq in physical_qubits:
        c = sum(
            1.0 / self.distance_matrix[pq][other]
            for other in physical_qubits
            if other != pq and self.distance_matrix[pq][other] not in (0, float('inf'))
        )
        pq_centrality[pq] = c

    pq_r2 = {}
    for pq in physical_qubits:
        nbrs = set(pq_adj[pq])
        for nb in pq_adj[pq]:
            nbrs.update(pq_adj[nb])
        nbrs.discard(pq)
        pq_r2[pq] = list(nbrs)

    # --- Step 5: Rearrangement-inequality Hungarian cost matrix ---
    lq_sorted_weights = {
        lq: sorted(lq_interactions[lq].values(), reverse=True)
        for lq in logical_qubits
    }
    pq_sorted_dists = {
        pq: sorted(
            self.distance_matrix[pq][other]
            for other in physical_qubits
            if other != pq and self.distance_matrix[pq][other] != float('inf')
        )
        for pq in physical_qubits
    }

    cost_matrix = np.zeros((n_lq, n_pq))
    for i, lq in enumerate(logical_qubits):
        weights = lq_sorted_weights[lq]
        if not weights:
            continue
        for j, pq in enumerate(physical_qubits):
            dists = pq_sorted_dists[pq]
            cost = 0.0
            for k, w in enumerate(weights):
                d = dists[k] if k < len(dists) else (dists[-1] + k - len(dists) + 1)
                cost += w * d
            cost_matrix[i, j] = cost

    row_ind, col_ind = linear_sum_assignment(cost_matrix)
    hungarian_map = {
        logical_qubits[r]: physical_qubits[c]
        for r, c in zip(row_ind, col_ind)
    }

    # --- Step 6: Greedy BFS seed ---
    def greedy_bfs_seed(anchor_pq=None, seed_pair=None):
        lq_phys = {}
        phys_used = set()

        if seed_pair is not None:
            slq1, slq2 = seed_pair
        elif interaction_weight:
            slq1, slq2 = max(interaction_weight, key=interaction_weight.__getitem__)
        else:
            slq1 = logical_qubits[0]
            slq2 = logical_qubits[1] if n_lq > 1 else logical_qubits[0]

        if anchor_pq is not None and anchor_pq in pq_adj:
            sp1 = anchor_pq
            sp2 = max(pq_adj[sp1], key=lambda p: pq_degree[p], default=sp1)
        else:
            best_score, sp1, sp2 = -1, physical_qubits[0], physical_qubits[min(1, n_pq - 1)]
            for p1 in physical_qubits:
                for p2 in pq_adj[p1]:
                    s = pq_degree[p1] + pq_degree[p2]
                    if s > best_score:
                        best_score, sp1, sp2 = s, p1, p2

        w1 = sum(lq_interactions[slq1].values())
        w2 = sum(lq_interactions[slq2].values())
        if (w1 >= w2) == (pq_degree[sp1] >= pq_degree[sp2]):
            lq_phys[slq1], lq_phys[slq2] = sp1, sp2
        else:
            lq_phys[slq1], lq_phys[slq2] = sp2, sp1

        phys_used.update(lq_phys.values())
        placed = set(lq_phys.keys())

        while len(placed) < n_lq:
            best_lq, best_score_lq = None, -1
            for lq in logical_qubits:
                if lq in placed:
                    continue
                s = sum(lq_interactions[lq].get(p, 0) for p in placed)
                if s > best_score_lq:
                    best_score_lq, best_lq = s, lq
            if best_lq is None:
                best_lq = next(lq for lq in logical_qubits if lq not in placed)

            best_pq, best_pq_cost = None, float('inf')
            for pq in physical_qubits:
                if pq in phys_used:
                    continue
                cost = sum(
                    w * self.distance_matrix[pq][lq_phys[nb]]
                    for nb, w in lq_interactions[best_lq].items()
                    if nb in placed
                )
                if cost < best_pq_cost:
                    best_pq_cost, best_pq = cost, pq
            if best_pq is None:
                best_pq = next(pq for pq in physical_qubits if pq not in phys_used)

            lq_phys[best_lq] = best_pq
            phys_used.add(best_pq)
            placed.add(best_lq)

        return lq_phys

    # --- Step 7: Build bijection ---
    def make_mapping(lq_phys):
        md = list(range(self.num_qubits))
        for lq, pq in lq_phys.items():
            md[lq] = pq
        assigned = set(lq_phys.values())
        remaining = [pq for pq in range(self.num_qubits) if pq not in assigned]
        ri = 0
        for lq in range(self.num_qubits):
            if lq not in lq_phys:
                md[lq] = remaining[ri]
                ri += 1
        rmd = list(range(self.num_qubits))
        for lq in range(self.num_qubits):
            rmd[md[lq]] = lq
        return md, rmd

    # --- Step 8: Cost utilities ---
    def qap_cost(md):
        return sum(
            w * self.distance_matrix[md[q1]][md[q2]]
            for (q1, q2), w in interaction_weight.items()
        )

    def swap_delta(p1, p2, md, rmd):
        a, b = rmd[p1], rmd[p2]
        delta = 0.0
        for nb, w in lq_interactions[a].items():
            pp = md[nb]
            if pp == p2:
                continue
            delta += w * (self.distance_matrix[p2][pp] - self.distance_matrix[p1][pp])
        for nb, w in lq_interactions[b].items():
            pp = md[nb]
            if pp == p1:
                continue
            delta += w * (self.distance_matrix[p1][pp] - self.distance_matrix[p2][pp])
        return delta

    def do_swap(p1, p2, md, rmd):
        a, b = rmd[p1], rmd[p2]
        md[a], md[b] = p2, p1
        rmd[p1], rmd[p2] = b, a

    lq_by_degree = sorted(
        logical_qubits,
        key=lambda lq: sum(lq_interactions[lq].values()),
        reverse=True
    )

    # --- Step 9: Local search with dynamic cost-guided ordering ---
    # NEW: each iteration re-ranks qubits by their marginal cost contribution,
    # so persistently mis-placed qubits get processed first.
    def local_search(md, rmd, max_iters=400, full_every=4):
        for iteration in range(max_iters):
            improved = False
            # Compute per-qubit cost for dynamic ordering
            qubit_costs = defaultdict(float)
            for (q1, q2), w in interaction_weight.items():
                c = w * self.distance_matrix[md[q1]][md[q2]]
                qubit_costs[q1] += c
                qubit_costs[q2] += c
            ordered = sorted(logical_qubits, key=lambda lq: qubit_costs[lq], reverse=True)

            do_full = (iteration % full_every == 0)
            for lq in ordered:
                p1 = md[lq]
                best_d, best_p2 = -1e-9, -1
                for p2 in pq_r2[p1]:
                    d = swap_delta(p1, p2, md, rmd)
                    if d < best_d:
                        best_d, best_p2 = d, p2
                if do_full:
                    r2_set = set(pq_r2[p1])
                    r2_set.add(p1)
                    for p2 in physical_qubits:
                        if p2 in r2_set:
                            continue
                        d = swap_delta(p1, p2, md, rmd)
                        if d < best_d:
                            best_d, best_p2 = d, p2
                if best_p2 != -1:
                    do_swap(p1, best_p2, md, rmd)
                    improved = True
            if not improved:
                break
        return md, rmd

    def local_search_full(md, rmd, max_iters=400):
        for _ in range(max_iters):
            improved = False
            for lq in lq_by_degree:
                p1 = md[lq]
                best_d, best_p2 = -1e-9, -1
                for p2 in physical_qubits:
                    if p2 == p1:
                        continue
                    d = swap_delta(p1, p2, md, rmd)
                    if d < best_d:
                        best_d, best_p2 = d, p2
                if best_p2 != -1:
                    do_swap(p1, best_p2, md, rmd)
                    improved = True
            if not improved:
                break
        return md, rmd

    # --- Step 10: Crossover operator (NEW) ---
    # Combines two good solutions by fixing the best-placed half from md1
    # and greedily completing using md2's preferences.
    def crossover(md1, rmd1, md2, rmd2):
        # Score each logical qubit by its cost contribution in md1
        scores = []
        for lq in logical_qubits:
            s = sum(
                w * self.distance_matrix[md1[lq]][md1[nb]]
                for nb, w in lq_interactions[lq].items()
            )
            scores.append((s, lq))
        scores.sort()  # lower cost = better placed

        n_keep = max(1, n_lq // 2)
        keep_lqs = set(lq for _, lq in scores[:n_keep])

        new_md = [None] * self.num_qubits
        used_pqs = set()
        for lq in keep_lqs:
            new_md[lq] = md1[lq]
            used_pqs.add(md1[lq])

        # Remaining: sort by interaction with kept qubits (place most-connected first)
        remaining_lqs = [lq for lq in logical_qubits if lq not in keep_lqs]
        remaining_lqs.sort(
            key=lambda lq: sum(lq_interactions[lq].get(klq, 0) for klq in keep_lqs),
            reverse=True
        )

        # Maintain placed_map incrementally
        placed_map = {lq: new_md[lq] for lq in keep_lqs}

        for lq in remaining_lqs:
            pq2 = md2[lq]
            if pq2 not in used_pqs:
                new_md[lq] = pq2
                used_pqs.add(pq2)
                placed_map[lq] = pq2
            else:
                best_pq, best_c = None, float('inf')
                for pq in physical_qubits:
                    if pq in used_pqs:
                        continue
                    c = sum(
                        w * self.distance_matrix[pq][placed_map[nb]]
                        for nb, w in lq_interactions[lq].items()
                        if nb in placed_map
                    )
                    if c < best_c:
                        best_c, best_pq = c, pq
                if best_pq is None:
                    best_pq = next(pq for pq in physical_qubits if pq not in used_pqs)
                new_md[lq] = best_pq
                used_pqs.add(best_pq)
                placed_map[lq] = best_pq

        # Assign non-logical qubits
        remaining_phys = sorted(set(range(self.num_qubits)) - used_pqs)
        ri = 0
        for q in range(self.num_qubits):
            if new_md[q] is None:
                new_md[q] = remaining_phys[ri]
                ri += 1
        new_rmd = list(range(self.num_qubits))
        for q in range(self.num_qubits):
            new_rmd[new_md[q]] = q
        return new_md, new_rmd

    # --- Step 11: Multi-start seeding ---
    top_central = sorted(physical_qubits, key=lambda p: pq_centrality[p], reverse=True)
    top_degree = sorted(physical_qubits, key=lambda p: pq_degree[p], reverse=True)
    sorted_interactions = sorted(interaction_weight.items(), key=lambda x: x[1], reverse=True)
    alt_pairs = [pair for pair, _ in sorted_interactions[1:3]] if len(sorted_interactions) > 1 else []

    seed_configs = [
        (hungarian_map, None),
        (None, (None, None)),
    ]
    for anchor in top_central[:2]:
        seed_configs.append((None, (anchor, None)))
    for anchor in top_degree[:2]:
        seed_configs.append((None, (anchor, None)))
    for pair in alt_pairs:
        seed_configs.append((None, (None, pair)))

    best_md, best_rmd, best_cost = None, None, float('inf')
    population = []

    for fixed_map, bfs_args in seed_configs:
        if fixed_map is not None:
            md, rmd = make_mapping(fixed_map)
        else:
            anchor_pq, seed_pair = bfs_args
            bfs_map = greedy_bfs_seed(anchor_pq=anchor_pq, seed_pair=seed_pair)
            md, rmd = make_mapping(bfs_map)
        md, rmd = local_search_full(md, rmd)
        c = qap_cost(md)
        population.append((c, md[:], rmd[:]))
        if c < best_cost:
            best_cost = c
            best_md, best_rmd = md[:], rmd[:]

    # --- Step 12: Crossover phase between top solutions (NEW) ---
    population.sort(key=lambda x: x[0])
    if len(population) >= 2:
        for i in range(min(3, len(population) - 1)):
            _, md1, rmd1 = population[i]
            _, md2, rmd2 = population[i + 1]
            md_c, rmd_c = crossover(md1, rmd1, md2, rmd2)
            md_c, rmd_c = local_search_full(md_c, rmd_c)
            c = qap_cost(md_c)
            population.append((c, md_c[:], rmd_c[:]))
            if c < best_cost:
                best_cost = c
                best_md, best_rmd = md_c[:], rmd_c[:]

    # --- Step 13: UCB1-adaptive ILS with 5 perturbation strategies (NEW) ---
    rng = random.Random(42)
    n_restarts = min(30, max(8, n_lq // 2))

    n_strategies = 5
    strategy_success = [0] * n_strategies
    strategy_attempts = [1] * n_strategies  # start at 1 to avoid log(0)

    for restart in range(n_restarts):
        md = best_md[:]
        rmd = best_rmd[:]

        # UCB1-style adaptive strategy selection (NEW)
        if restart < n_strategies:
            strategy = restart
        else:
            rates = [strategy_success[i] / strategy_attempts[i] for i in range(n_strategies)]
            ucb = [
                rates[i] + 0.15 * math.sqrt(math.log(restart + 1) / strategy_attempts[i])
                for i in range(n_strategies)
            ]
            strategy = max(range(n_strategies), key=lambda i: ucb[i])

        strategy_attempts[strategy] += 1

        if strategy == 0:
            # Rotation permutation
            k = rng.randint(max(3, n_lq // 4), max(4, n_lq // 2 + 1))
            k = min(k, n_lq)
            sample = rng.sample(logical_qubits, k)
            positions = [md[lq] for lq in sample]
            offset = rng.randint(1, k - 1)
            rotated = positions[offset:] + positions[:offset]
            for lq, tgt in zip(sample, rotated):
                cur = md[lq]
                if cur != tgt:
                    do_swap(cur, tgt, md, rmd)

        elif strategy == 1:
            # Random multi-swap
            n_swaps = rng.randint(2, max(3, n_lq // 3))
            pool = rng.sample(logical_qubits, min(n_swaps * 2, n_lq))
            for i in range(0, len(pool) - 1, 2):
                do_swap(md[pool[i]], md[pool[i + 1]], md, rmd)

        elif strategy == 2:
            # Cost-guided perturbation: displace worst-placed interacting pairs
            worst = sorted(
                interaction_weight.items(),
                key=lambda x: x[1] * self.distance_matrix[md[x[0][0]]][md[x[0][1]]],
                reverse=True
            )[:max(2, n_lq // 4)]
            perturbed = set()
            for (q1, q2), _ in worst:
                if q1 not in perturbed and q2 not in perturbed:
                    candidates = [lq for lq in logical_qubits
                                  if lq not in perturbed and lq != q1 and lq != q2]
                    if candidates:
                        q3 = rng.choice(candidates)
                        do_swap(md[q2], md[q3], md, rmd)
                        perturbed.update([q1, q2, q3])

        elif strategy == 3:
            # Subgraph relocation
            seed_lq = rng.choice(lq_by_degree[:max(1, n_lq // 4)])
            sub_size = rng.randint(2, max(3, n_lq // 5))
            subgraph = [seed_lq]
            lq_frontier = sorted(
                lq_interactions[seed_lq].keys(),
                key=lambda q: lq_interactions[seed_lq][q],
                reverse=True
            )
            for lq in lq_frontier:
                if len(subgraph) >= sub_size:
                    break
                if lq not in subgraph:
                    subgraph.append(lq)

            cand_anchors = top_central[:max(3, n_pq // 6)]
            new_anchor = rng.choice(cand_anchors)
            phys_blocked = set(md[lq] for lq in logical_qubits if lq not in subgraph)
            new_positions = []
            phys_frontier = [new_anchor]
            visited_pq = {new_anchor}
            for pq in phys_frontier:
                if pq not in phys_blocked:
                    new_positions.append(pq)
                if len(new_positions) >= len(subgraph):
                    break
                for nb in pq_adj[pq]:
                    if nb not in visited_pq:
                        visited_pq.add(nb)
                        phys_frontier.append(nb)

            if len(new_positions) < len(subgraph):
                for pq in physical_qubits:
                    if pq not in phys_blocked and pq not in set(new_positions):
                        new_positions.append(pq)
                        if len(new_positions) >= len(subgraph):
                            break

            for i, lq in enumerate(subgraph):
                if i < len(new_positions):
                    tgt = new_positions[i]
                    cur = md[lq]
                    if cur != tgt:
                        do_swap(cur, tgt, md, rmd)

        else:  # strategy == 4 — NEW: chain reversal along interaction graph
            # Build a chain by following highest-weight edges, then reverse
            # physical positions (2-opt-style). Correct because sequential
            # do_swap calls update md/rmd in-place.
            chain_len = rng.randint(3, min(7, n_lq))
            start_lq = rng.choice(lq_by_degree[:max(2, n_lq // 3)])
            chain = [start_lq]
            visited_lq = {start_lq}
            for _ in range(chain_len - 1):
                last = chain[-1]
                neighbors = [
                    (nb, w) for nb, w in lq_interactions[last].items()
                    if nb not in visited_lq
                ]
                if not neighbors:
                    break
                next_lq = max(neighbors, key=lambda x: x[1])[0]
                chain.append(next_lq)
                visited_lq.add(next_lq)

            if len(chain) >= 2:
                positions = [md[lq] for lq in chain]
                positions_rev = positions[::-1]
                for lq, tgt in zip(chain, positions_rev):
                    cur = md[lq]
                    if cur != tgt:
                        do_swap(cur, tgt, md, rmd)

        md, rmd = local_search(md, rmd)
        c = qap_cost(md)
        if c < best_cost:
            best_cost = c
            best_md = md[:]
            best_rmd = rmd[:]
            strategy_success[strategy] += 1

    # --- Step 14: SA with periodic reheat when stuck (NEW) ---
    md = best_md[:]
    rmd = best_rmd[:]
    current_cost = best_cost

    active_pqs = list(set(md[lq] for lq in logical_qubits if lq_interactions[lq]))
    if len(active_pqs) < 2:
        active_pqs = physical_qubits[:]

    n_sa_iters = max(4000, n_lq * 350)
    T_start = max(current_cost * 0.05, 1.0)
    T_end = max(current_cost * 0.00004, 1e-5)
    alpha = (T_end / T_start) ** (1.0 / n_sa_iters)
    T = T_start

    phase_len = max(500, n_sa_iters // 10)
    n_accepted_phase = 0

    for i in range(n_sa_iters):
        p1, p2 = rng.sample(active_pqs, 2)
        delta = swap_delta(p1, p2, md, rmd)
        if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-delta / T)):
            do_swap(p1, p2, md, rmd)
            current_cost += delta
            n_accepted_phase += 1
            if current_cost < best_cost:
                best_cost = current_cost
                best_md = md[:]
                best_rmd = rmd[:]
        T *= alpha

        # Periodic reheat: reset to best and warm up if acceptance collapses (NEW)
        if (i + 1) % phase_len == 0:
            accept_rate = n_accepted_phase / phase_len
            n_accepted_phase = 0
            if accept_rate < 0.002:
                md = best_md[:]
                rmd = best_rmd[:]
                current_cost = best_cost
                T = max(T, T_start * 0.04)

    # --- Step 15: Final full local search ---
    best_md, best_rmd = local_search_full(best_md, best_rmd, max_iters=500)

    self.mapping_dict = best_md
    self.reverse_mapping_dict = best_rmd

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Key changes from Round 6 summarized:**

| Feature | Round 6 | Round 7 |
|---|---|---|
| Local search qubit order | Static `lq_by_degree` | Dynamic per-iteration cost-ranking |
| Multi-start population | Discarded after seeding | Kept; top pairs are crossed over |
| Crossover | None | New: fix best-placed half from solution A, fill gaps using solution B's preferences |
| ILS strategy selection | Fixed `restart % 4` | UCB1-adaptive (5 strategies) |
| 5th perturbation | None | Chain reversal along interaction graph (2-opt-style) |
| SA | Single-phase geometric | + Periodic reheat when acceptance rate < 0.2% |
| SA iterations | `n_lq * 300` | `n_lq * 350` |
