```python
                for lq in crossover_lqs:
                    target_pq = other_md[lq]
                    cur_pq = md[lq]
                    if cur_pq != target_pq:
                        do_swap(cur_pq, target_pq, md, rmd)

        md, rmd = local_search(md, rmd)
        # Every 4th restart, apply 3-cycle search to break out of deeper optima
        if restart % 4 == 0:
            md, rmd = local_search_3cycle(md, rmd, max_iters=25)
            md, rmd = local_search(md, rmd, max_iters=100)
        c = qap_cost(md)
        if c < best_cost:
            best_cost = c
            best_md = md[:]
            best_rmd = rmd[:]
        if len(population) < 5 or c < population[-1][0]:
            population.append((c, md[:], rmd[:]))
            population = sorted(population, key=lambda x: x[0])[:5]

    # --- Step 12: SA refinement with reheating ---
    md = best_md[:]
    rmd = best_rmd[:]
    current_cost = best_cost

    active_pqs = list(set(md[lq] for lq in logical_qubits if lq_combined[lq]))
    if len(active_pqs) < 2:
        active_pqs = physical_qubits

    n_sa_iters = max(4000, n_lq * 380)
    T_start = max(current_cost * 0.05, 0.5)
    T_end = max(current_cost * 0.000035, 1e-4)
    alpha = (T_end / T_start) ** (1.0 / n_sa_iters)
    T = T_start

    no_improve_streak = 0
    reheat_interval = n_sa_iters // 6
    last_improvement_iter = 0

    for sa_iter in range(n_sa_iters):
        p1, p2 = rng.sample(active_pqs, 2)
        delta = swap_delta(p1, p2, md, rmd)
        if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-delta / T)):
            do_swap(p1, p2, md, rmd)
            current_cost += delta
            if current_cost < best_cost:
                best_cost = current_cost
                best_md = md[:]
                best_rmd = rmd[:]
                no_improve_streak = 0
                last_improvement_iter = sa_iter
            else:
                no_improve_streak += 1
        else:
            no_improve_streak += 1

        T *= alpha

        if no_improve_streak >= reheat_interval:
            T = max(T, T_start * 0.08)
            no_improve_streak = 0

    # --- Step 13: Final local search + 3-cycle on best solution ---
    best_md, best_rmd = local_search_full(best_md, best_rmd, max_iters=500)
    best_md, best_rmd = local_search_3cycle(best_md, best_rmd, max_iters=60)
    best_md, best_rmd = local_search_full(best_md, best_rmd, max_iters=200)

    self.mapping_dict = best_md
    self.reverse_mapping_dict = best_rmd

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

**Summary of changes from Round 9 → Round 10:**

| Change | Rationale |
|---|---|
| **Spectral seed** (Step 5b) | Uses criticality-weighted graph Laplacian eigenvectors for both logical and physical graphs, matched via Hungarian. This captures global graph structure that the BFS/Hungarian seeds miss. |
| **3-cycle local search** (Step 9b) | Tries all 3-way cyclic rotations `lq1→p2, lq2→p3, lq3→p1` and its reverse. Standard 2-swap local search cannot improve if the best move involves simultaneously relocating 3 qubits — this addresses that. Applied after each initial seed and every 4th ILS restart. |
| **Spectral seed in multi-start** (Step 10) | Adds `spec_map` as a third deterministic seed alongside Hungarian and BFS. |
| **3-cycle after initial seeds** | Each seed solution gets `local_search_full` → `local_search_3cycle(40)` → `local_search_full(150)` to reach a deeper local optimum. |
| **3-cycle in final post-processing** | After SA, the final sequence is `local_search_full(500)` → `local_search_3cycle(60)` → `local_search_full(200)`, ensuring 3-cycle optima. |
| **Minor SA tuning** | `n_lq * 380` iters (vs 350), slightly lower `T_end` multiplier (`0.000035` vs `0.00004`). |
