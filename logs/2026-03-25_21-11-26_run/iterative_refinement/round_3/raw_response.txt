```python
                    break
                triple = rng.sample(candidates, 3)
                p0, p1_t, p2_t = md[triple[0]], md[triple[1]], md[triple[2]]
                do_swap(p0, p1_t, md, rmd)
                do_swap(p1_t, p2_t, md, rmd)

        else:
            # Strategy 6: Front-layer focused perturbation
            early_pairs = [(pair, w) for pair, w in interaction_weight.items() if w > 0]
            early_pairs.sort(key=lambda x: x[1], reverse=True)
            n_relocate = rng.randint(1, max(2, min(4, len(early_pairs))))
            moved = set()
            for (q1, q2), _ in early_pairs[:n_relocate]:
                if q1 in moved or q2 in moved:
                    continue
                best_c = float('inf')
                best_p1, best_p2 = md[q1], md[q2]
                for pp1 in top_central[:max(6, n_pq // 5)]:
                    for pp2 in pq_adj[pp1]:
                        c = 0.0
                        for nb, w in lq_combined[q1].items():
                            pnb = pp2 if nb == q2 else md[nb]
                            c += w * self.distance_matrix[pp1][pnb]
                        for nb, w in lq_combined[q2].items():
                            pnb = pp1 if nb == q1 else md[nb]
                            c += w * self.distance_matrix[pp2][pnb]
                        if c < best_c:
                            best_c = c
                            best_p1, best_p2 = pp1, pp2
                cur_p1 = md[q1]
                if best_p1 != cur_p1:
                    do_swap(cur_p1, best_p1, md, rmd)
                cur_p2 = md[q2]
                if best_p2 != cur_p2:
                    do_swap(cur_p2, best_p2, md, rmd)
                moved.update([q1, q2])

        md, rmd = local_search(md, rmd)
        c = qap_cost(md)
        if c < best_cost:
            best_cost = c
            best_md = md[:]
            best_rmd = rmd[:]
        if len(population) < 6 or c < population[-1][0]:
            population.append((c, md[:], rmd[:]))
            population = sorted(population, key=lambda x: x[0])[:6]

    # --- Step 12: Extended SA with biased neighborhood + warm restarts ---
    md = best_md[:]
    rmd = best_rmd[:]
    current_cost = best_cost

    active_pqs = list(set(md[lq] for lq in logical_qubits if lq_combined[lq]))
    if len(active_pqs) < 2:
        active_pqs = physical_qubits

    # Build biased sampling for SA
    interaction_pairs_phys = []
    interaction_pairs_weights = []
    for (q1, q2), w in combined_weight.items():
        interaction_pairs_phys.append((q1, q2))
        interaction_pairs_weights.append(w)

    total_iw = sum(interaction_pairs_weights) if interaction_pairs_weights else 1.0
    interaction_pairs_cum = []
    cum = 0.0
    for w in interaction_pairs_weights:
        cum += w / total_iw
        interaction_pairs_cum.append(cum)

    def sample_biased_swap(md, rng_local):
        if interaction_pairs_phys and rng_local.random() < 0.7:
            r = rng_local.random()
            idx = 0
            for idx in range(len(interaction_pairs_cum)):
                if interaction_pairs_cum[idx] >= r:
                    break
            q1, q2 = interaction_pairs_phys[idx]
            p1 = md[q1]
            if rng_local.random() < 0.5 and pq_adj[p1]:
                p2 = rng_local.choice(pq_adj[p1])
            else:
                p2 = rng_local.choice(active_pqs)
                while p2 == p1 and len(active_pqs) > 1:
                    p2 = rng_local.choice(active_pqs)
            return p1, p2
        else:
            return rng_local.sample(active_pqs, 2)

    n_sa_iters = max(6000, n_lq * 500)
    T_start = max(current_cost * 0.06, 0.5)
    T_end = max(current_cost * 0.00003, 1e-4)
    alpha = (T_end / T_start) ** (1.0 / n_sa_iters)
    T = T_start

    no_improve_streak = 0
    reheat_interval = n_sa_iters // 7
    n_reheats = 0
    max_reheats = 5

    for sa_iter in range(n_sa_iters):
        p1, p2 = sample_biased_swap(md, rng)
        if p1 == p2:
            continue
        delta = swap_delta(p1, p2, md, rmd)
        if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-delta / T)):
            do_swap(p1, p2, md, rmd)
            current_cost += delta
            if current_cost < best_cost:
                best_cost = current_cost
                best_md = md[:]
                best_rmd = rmd[:]
                no_improve_streak = 0
            else:
                no_improve_streak += 1
        else:
            no_improve_streak += 1

        T *= alpha

        if no_improve_streak >= reheat_interval and n_reheats < max_reheats:
            md = best_md[:]
            rmd = best_rmd[:]
            current_cost = best_cost
            T = T_start * (0.2 ** (n_reheats + 1))
            no_improve_streak = 0
            n_reheats += 1

    # --- Step 13: Final full local search ---
    best_md, best_rmd = local_search_full(best_md, best_rmd, max_iters=600)

    self.mapping_dict = best_md
    self.reverse_mapping_dict = best_rmd

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)
```

## Summary of Changes from v2 (306.62)

**1. DAG-aware dependency weighting** (most impactful change):
- Builds a 2Q-gate DAG and computes transitive successor counts via topological-order DP
- Each gate's weight is boosted by `1.0 + 0.5 * log(1 + deps)` where `deps` is the number of downstream 2Q gates
- This directly aligns the mapper's objective with the router's scoring function (`(deps+1) * distance`), so gates that are bottlenecks for the rest of the circuit get properly prioritized

**2. Per-axis spectral sign optimization**:
- Old: tried 2 sign orientations (global flip vs no flip)
- New: tries all 2^k = 16 per-axis sign combinations with vectorized cost computation
- Eigenvectors have arbitrary sign per axis; optimizing independently finds better spectral-to-physical alignment

**3. Consistent cost functions**:
- Old: `qap_cost` used `interaction_weight` but `swap_delta` used `lq_combined` (includes second-order terms)
- New: both use `combined_weight`/`lq_combined` consistently, so SA and local search optimize the same objective

**4. Biased SA neighborhood sampling**:
- 70% of SA moves sample a high-weight interaction pair and swap one endpoint with a neighbor or active qubit
- 30% remain uniform random for diversity
- This concentrates SA effort on swaps that actually affect cost, rather than wasting iterations on irrelevant qubit pairs

**5. More ILS restarts (48 vs 40) with 7th perturbation strategy**:
- New strategy 6: front-layer focused perturbation that relocates the highest-weight interaction pairs to adjacent physical positions

**6. More SA iterations** (n_lq * 500 vs 400) and 5 reheats vs 4, with adjusted reheat factor (0.2 vs 0.15)
