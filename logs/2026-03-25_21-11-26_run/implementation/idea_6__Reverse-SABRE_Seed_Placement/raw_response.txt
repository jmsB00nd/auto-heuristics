def init_mapping(self):
    from collections import deque

    num_trials = 5

    # Build DAG for reverse traversal (only need 2q gate info)
    successors2q, predecessors2q, successors_full, predecessors_full, access2q = generate_dag(
        self.access, self.write_dict, self.num_qubits,
        enforce_read_after_read=True, transitive_reduction=False
    )

    # Compute reverse topological order of 2-qubit gates
    # (We process gates from outputs back to inputs)
    in_degree = {g: len(successors2q[g]) for g in successors2q}
    # Start from gates with no successors (output gates of the 2q DAG)
    queue = deque([g for g in successors2q if in_degree[g] == 0])
    reverse_topo_order = []
    while queue:
        g = queue.popleft()
        reverse_topo_order.append(g)
        for pred in predecessors2q.get(g, []):
            in_degree[pred] -= 1
            if in_degree[pred] == 0:
                queue.append(pred)

    best_mapping = None
    best_reverse_mapping = None
    best_cost = float('inf')

    for _ in range(num_trials):
        # Step 1: Random initial mapping
        mapping, reverse_mapping = generate_random_initial_mapping(self.num_qubits)

        # Step 2-3: Process gates in reverse topological order
        for gate in reverse_topo_order:
            qubits = access2q.get(gate)
            if qubits is None or len(qubits) != 2:
                continue

            q1, q2 = qubits
            phys_q1 = mapping[q1]
            phys_q2 = mapping[q2]

            # If already adjacent, skip
            if (phys_q1, phys_q2) in self.backend_connections or \
               (phys_q2, phys_q1) in self.backend_connections:
                continue

            # Find shortest path from phys_q1 to phys_q2 via BFS
            parent = {phys_q1: None}
            bfs_queue = deque([phys_q1])
            while bfs_queue:
                curr = bfs_queue.popleft()
                if curr == phys_q2:
                    break
                for neighbor in self.backend[curr]:
                    if neighbor not in parent:
                        parent[neighbor] = curr
                        bfs_queue.append(neighbor)

            # Reconstruct path
            path = []
            node = phys_q2
            while node is not None:
                path.append(node)
                node = parent[node]
            path.reverse()

            # Apply SWAPs along path to bring phys_q1 toward phys_q2
            # Swap along the first edge of the path to move q1 closer
            if len(path) >= 2:
                swap_pair = (path[0], path[1])
                swap_logical_physical_mappings(
                    mapping, reverse_mapping, swap_pair, inplace=True
                )

        # Step 4-6: Evaluate this mapping's quality
        cost = 0
        for gate in access2q:
            q1, q2 = access2q[gate]
            cost += self.distance_matrix[mapping[q1]][mapping[q2]]

        if cost < best_cost:
            best_cost = cost
            best_mapping = mapping[:]
            best_reverse_mapping = reverse_mapping[:]

    # Set the winning mapping
    self.mapping_dict = best_mapping
    self.reverse_mapping_dict = best_reverse_mapping

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

This implements Reverse-SABRE Seed Placement:

1. **Reverse topological order** - computed by starting from sink nodes (no successors) in the 2-qubit DAG and working backward through predecessors
2. **Reverse simulation** - for each 2-qubit gate in reverse order, if its logical qubits aren't adjacent under the current mapping, BFS finds the shortest path and a single SWAP is applied along the first edge to move them closer
3. **Multi-trial** - runs 5 trials with different random seeds, keeping the mapping with the lowest total distance cost across all 2-qubit gates
4. **Bijectivity preserved** - starts from a valid permutation (`generate_random_initial_mapping`) and only applies SWAPs (bijective), so the mapping stays valid throughout
