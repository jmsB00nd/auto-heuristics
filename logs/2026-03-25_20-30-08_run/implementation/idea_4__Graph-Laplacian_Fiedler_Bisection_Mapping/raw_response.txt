def init_mapping(self):
    import numpy as np
    from collections import defaultdict

    # ── 1. Collect logical qubits and build weighted interaction graph ──
    logical_qubits = sorted({q for qubits in self.access.values() for q in qubits})
    num_logical = len(logical_qubits)

    if num_logical == 0:
        self.mapping_dict = {}
        self.reverse_mapping_dict = {}
        return

    # Build weighted adjacency for logical interaction graph
    log_adj = defaultdict(lambda: defaultdict(float))
    for gate, qubits in self.access.items():
        if len(qubits) == 2:
            q1, q2 = qubits
            log_adj[q1][q2] += 1.0
            log_adj[q2][q1] += 1.0

    # Physical qubits
    phys_qubits = sorted(self.backend.keys())

    def fiedler_partition(nodes, adj):
        """Partition nodes using the Fiedler vector of the graph Laplacian."""
        n = len(nodes)
        if n <= 1:
            return nodes, []

        idx = {v: i for i, v in enumerate(nodes)}
        L = np.zeros((n, n))
        for u in nodes:
            for v, w in adj[u].items():
                if v in idx:
                    i, j = idx[u], idx[v]
                    L[i, j] -= w
                    L[j, i] -= w
                    L[i, i] += w
                    L[j, j] += w
        # Avoid double-counting: we visited each edge from both sides
        L /= 2.0

        # Compute eigenvalues/vectors of the Laplacian
        eigenvalues, eigenvectors = np.linalg.eigh(L)

        # Find the Fiedler vector (second-smallest eigenvalue)
        # Skip near-zero eigenvalues (the trivial ones)
        fiedler_idx = 1 if n > 1 else 0
        fiedler_vec = eigenvectors[:, fiedler_idx]

        group_a = [nodes[i] for i in range(n) if fiedler_vec[i] < 0]
        group_b = [nodes[i] for i in range(n) if fiedler_vec[i] >= 0]

        # Ensure non-empty partitions
        if len(group_a) == 0:
            group_a = [group_b.pop(0)]
        elif len(group_b) == 0:
            group_b = [group_a.pop(0)]

        return group_a, group_b

    def hw_fiedler_partition(nodes):
        """Partition hardware nodes using Fiedler vector with unweighted adjacency."""
        n = len(nodes)
        if n <= 1:
            return nodes, []

        node_set = set(nodes)
        idx = {v: i for i, v in enumerate(nodes)}
        L = np.zeros((n, n))
        for u in nodes:
            for v in self.backend[u]:
                if v in node_set:
                    i, j = idx[u], idx[v]
                    L[i, j] = -1
                    L[j, i] = -1
                    L[i, i] += 1
                    L[j, j] += 1
        L /= 2.0

        eigenvalues, eigenvectors = np.linalg.eigh(L)
        fiedler_vec = eigenvectors[:, 1] if n > 1 else eigenvectors[:, 0]

        group_a = [nodes[i] for i in range(n) if fiedler_vec[i] < 0]
        group_b = [nodes[i] for i in range(n) if fiedler_vec[i] >= 0]

        if len(group_a) == 0:
            group_a = [group_b.pop(0)]
        elif len(group_b) == 0:
            group_b = [group_a.pop(0)]

        return group_a, group_b

    def total_weight(nodes, adj):
        """Sum of internal edge weights for a set of nodes."""
        node_set = set(nodes)
        w = 0.0
        for u in nodes:
            for v, wt in adj[u].items():
                if v in node_set and v > u:
                    w += wt
        return w

    def total_degree_hw(nodes):
        """Sum of internal degrees for hardware subgraph."""
        node_set = set(nodes)
        d = 0
        for u in nodes:
            for v in self.backend[u]:
                if v in node_set:
                    d += 1
        return d

    def recursive_bisect(log_nodes, phys_nodes):
        """Recursively bisect and match logical to physical qubits."""
        if len(log_nodes) == 1:
            # Base case: assign the single logical qubit to a physical qubit
            return [(log_nodes[0], phys_nodes[0])]

        if len(log_nodes) == 0:
            return []

        # Partition logical qubits
        log_a, log_b = fiedler_partition(log_nodes, log_adj)

        # Partition physical qubits
        phys_a, phys_b = hw_fiedler_partition(phys_nodes)

        # Balance partition sizes to match logical splits
        # We need |phys_a| >= |log_a| and |phys_b| >= |log_b|
        target_a = len(log_a)
        target_b = len(log_b)

        # Rebalance physical partitions to match logical sizes
        # Move nodes between phys_a and phys_b so sizes fit
        while len(phys_a) < target_a:
            phys_a.append(phys_b.pop())
        while len(phys_b) < target_b:
            phys_b.append(phys_a.pop())

        # Match denser logical half to more connected hardware half
        w_log_a = total_weight(log_a, log_adj)
        w_log_b = total_weight(log_b, log_adj)
        d_phys_a = total_degree_hw(phys_a)
        d_phys_b = total_degree_hw(phys_b)

        # Denser logical group should go to more connected physical group
        # Check if current assignment is correct or needs swapping
        log_dense = log_a if w_log_a >= w_log_b else log_b
        log_sparse = log_b if w_log_a >= w_log_b else log_a
        phys_dense = phys_a if d_phys_a >= d_phys_b else phys_b
        phys_sparse = phys_b if d_phys_a >= d_phys_b else phys_a

        assignments = []
        assignments.extend(recursive_bisect(log_dense, phys_dense))
        assignments.extend(recursive_bisect(log_sparse, phys_sparse))
        return assignments

    # ── 2. Run recursive bisection ──────────────────────────────────────
    assignments = recursive_bisect(logical_qubits, phys_qubits)

    # ── 3. Populate mapping dictionaries ────────────────────────────────
    self.mapping_dict = {}
    self.reverse_mapping_dict = {}
    for lq, pq in assignments:
        self.mapping_dict[lq] = pq
        self.reverse_mapping_dict[pq] = lq

    if self.use_isl:
        self.isl_mapping = dict_to_isl_map(self.mapping_dict)

The algorithm:

1. **Builds a weighted logical interaction graph** from `self.access` (gate count between qubit pairs).
2. **Computes the Fiedler vector** (eigenvector of 2nd-smallest eigenvalue of the Laplacian) for both the logical and hardware graphs to partition nodes by sign.
3. **Matches the denser logical half to the more connected physical half** based on internal edge weight/degree.
4. **Rebalances physical partitions** so each has enough physical qubits for the corresponding logical partition.
5. **Recurses** until single qubits remain, where direct assignment is made.
6. **Collision-free by construction**: disjoint logical subsets are always assigned to disjoint physical subsets at every recursion level.
